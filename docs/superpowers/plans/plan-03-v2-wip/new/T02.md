const { BYOK_API_KEYS } = require("../../../src/config/secretKeys");

// An in-memory stand-in for EnvironmentManager's credential surface: every
// getter the credential authority reads plus an awaitable persistSecret whose
// timing and failure a test controls through persistImpl. It records only the
// names it persisted, never the values.
const GETTERS = {
  ...Object.fromEntries(BYOK_API_KEYS.map((k) => [k.get, k.env])),
  getCortiClientId: "CORTI_CLIENT_ID",
  getCortiClientSecret: "CORTI_CLIENT_SECRET",
  getCustomTranscriptionKey: "CUSTOM_TRANSCRIPTION_API_KEY",
  getCleanupCustomKey: "CUSTOM_CLEANUP_API_KEY",
  getBedrockAccessKeyId: "BEDROCK_ACCESS_KEY_ID",
  getBedrockSecretAccessKey: "BEDROCK_SECRET_ACCESS_KEY",
  getBedrockSessionToken: "BEDROCK_SESSION_TOKEN",
  getBedrockProfile: "BEDROCK_PROFILE",
  getAzureApiKey: "AZURE_OPENAI_API_KEY",
  getVertexApiKey: "VERTEX_API_KEY",
  getVertexProject: "VERTEX_PROJECT",
};

function fakeCredentialEnv(initial = {}) {
  const values = new Map(Object.entries(initial));
  const env = { values, persisted: [], persistImpl: null };
  for (const [getter, name] of Object.entries(GETTERS)) env[getter] = () => values.get(name) || "";
  env.persistSecret = async (name, value) => {
    env.persisted.push(name);
    if (env.persistImpl) await env.persistImpl(name, value);
    if (value) values.set(name, value);
    else values.delete(name);
  };
  return env;
}

// fs double for Codex presence: stat answers from `size` (null = no file) and
// any attempt to read the credential file fails the test.
function fakeCodexFs({ size = null } = {}) {
  const calls = [];
  return {
    calls,
    statSync(file) {
      calls.push(file);
      if (size === null) throw Object.assign(new Error("ENOENT"), { code: "ENOENT" });
      return { isFile: () => true, size };
    },
    readFileSync() {
      throw new Error("credential presence must never read auth.json");
    },
    promises: {
      readFile() {
        throw new Error("credential presence must never read auth.json");
      },
    },
  };
}

module.exports = { fakeCredentialEnv, fakeCodexFs };

   function createCredentialAuthority({
     env,
     codexHome,
     fs,
     broadcast = () => {},
     now = Date.now,
     undoMs = UNDO_MS,
     setTimer = (fn, ms) => setTimeout(fn, ms).unref?.(),
     randomToken = () => crypto.randomBytes(18).toString("base64url"),
   }) {
     const generations = new Map();
     const queues = new Map();
     const undoEntries = new Map(); // token → { id, generation, values | null, expiresAt }
     const listeners = new Set();

     const generation = (id) => generations.get(id) || 0;
     const bump = (id) => {
       const next = generation(id) + 1;
       generations.set(id, next);
       return next;
     };
     const fieldsOf = (id) =>
       Object.fromEntries(
         Object.entries(SLOTS[id].fields).map(([name, f]) => [name, !!env[f.get]()])
       );
     const present = (id) => {
       if (id === "codex") return hasStoredAccount({ fs, codexHome });
       if (!SLOTS[id]) return false;
       const stored = fieldsOf(id);
       return SLOTS[id].present
         ? !!SLOTS[id].present(stored, env)
         : Object.values(stored).some(Boolean);
     };
     const snapshot = (id) => ({
       id,
       present: present(id),
       generation: generation(id),
       fields: id === "codex" ? {} : fieldsOf(id),
     });
     const emit = (id, cause) => {
       const change = snapshot(id);
       broadcast("credentials-changed", change);
       for (const listener of listeners) {
         try {
           listener({ ...change, cause });
         } catch {}
       }
     };
     // One write at a time per identity, in call order, so a replacement save
     // can never be overtaken by the save it replaces.
     const enqueue = (id, task) => {
       const run = (queues.get(id) || Promise.resolve()).then(task);
       queues.set(id, run.catch(() => {}));
       return run;
     };
     const persist = async (id, values) => {
       for (const [name, value] of Object.entries(values)) {
         await env.persistSecret(SLOTS[id].fields[name].env, value);
       }
     };
     const fail = (error) => ({ ok: false, error });
     const slotError = (id) => (id === "codex" ? "unsupported" : "unknownCredential");

     function save(id, fieldName, value) {
       if (!SLOTS[id]) return Promise.resolve(fail(slotError(id)));
       if (!SLOTS[id].fields[fieldName]) return Promise.resolve(fail("unknownField"));
       if (typeof value !== "string" || value.length > MAX_SECRET_LENGTH) {
         return Promise.resolve(fail("invalidValue"));
       }
       // The renderer echoes the sentinel back for an unchanged key; never store it.
       if (value === STORED_SENTINEL) return Promise.resolve({ ok: true, generation: generation(id) });
       return enqueue(id, async () => {
         try {
           await persist(id, { [fieldName]: value.trim() });
         } catch {
           return fail("persistFailed");
         }
         const next = bump(id);
         emit(id, "save");
         return { ok: true, generation: next };
       });
     }

     function remove(id) {
       if (!SLOTS[id]) return Promise.resolve(fail(slotError(id)));
       return enqueue(id, async () => {
         const values = {};
         for (const [name, f] of Object.entries(SLOTS[id].fields)) {
           const stored = env[f.get]();
           if (stored) values[name] = stored;
         }
         if (Object.keys(values).length === 0) return fail("notStored");
         const cleared = {};
         try {
           for (const name of Object.keys(values)) {
             await env.persistSecret(SLOTS[id].fields[name].env, "");
             cleared[name] = values[name];
           }
         } catch {
           await persist(id, cleared).catch(() => {});
           return fail("persistFailed");
         }
         const next = bump(id);
         const undoToken = randomToken();
         const expiresAt = now() + undoMs;
         const entry = { id, generation: next, values, expiresAt };
         undoEntries.set(undoToken, entry);
         // The removed value must not outlive the Undo window in memory.
         setTimer(() => {
           entry.values = null;
         }, undoMs);
         emit(id, "remove");
         return { ok: true, undoToken, expiresAt, generation: next };
       });
     }

     function undoRemove(token) {
       const entry = typeof token === "string" ? undoEntries.get(token) : undefined;
       if (!entry) return Promise.resolve({ ok: false, reason: "unknown" });
       return enqueue(entry.id, async () => {
         if (undoEntries.get(token) !== entry) return { ok: false, reason: "unknown" };
         if (!entry.values || now() >= entry.expiresAt) {
           entry.values = null;
           return { ok: false, reason: "expired" };
         }
         if (generation(entry.id) !== entry.generation) {
           entry.values = null;
           return { ok: false, reason: "replaced" };
         }
         try {
           await persist(entry.id, entry.values);
         } catch {
           return { ok: false, reason: "persistFailed" };
         }
         undoEntries.delete(token);
         entry.values = null;
         const next = bump(entry.id);
         emit(entry.id, "undo");
         return { ok: true, generation: next };
       });
     }

     function reset() {
       for (const entry of undoEntries.values()) entry.values = null;
       undoEntries.clear();
       for (const id of [...Object.keys(SLOTS), "codex"]) {
         bump(id);
         emit(id, "reset");
       }
     }

     function refresh(id) {
       if (id === "codex" || SLOTS[id]) emit(id, "refresh");
     }

     return {
       list: () => CREDENTIAL_IDS.map(snapshot),
       present,
       generation,
       save,
       remove,
       undoRemove,
       reset,
       refresh,
       subscribe(listener) {
         listeners.add(listener);
         return () => listeners.delete(listener);
       },
     };
   }

   module.exports = {
     UNDO_MS,
     CREDENTIAL_IDS,
     byokSlotId,
     slotForCredentialRef,
     hasStoredAccount,
     createCredentialAuthority,
   };

       const [id, field] = SECRET_SLOTS[storeKey];
       let result: CredentialSaveResult;
       try {
         result = await credentials.save(id, field, value);
       } catch {
         result = { ok: false, error: "persistFailed" };
       }
       if (result.ok === false) {
         logger.warn("Failed to persist secret", { storeKey, error: result.error }, "settings");
         return { ok: false, error: result.error };
       }
       if (latestSecretSave[storeKey] === sequence && value !== "__stored__") {
         useSettingsStore.setState({
           [storeKey]: value.trim() ? "__stored__" : "",
         } as Partial<SettingsState>);
       }
       // Main already persisted (persistSecret); no .env rewrite from here.
       invalidateApiKeyCaches(SECRET_CACHE[storeKey], { persistEnv: false });
       return { ok: true };
     })();
     pendingSecretSaves.add(run);
     void run.finally(() => pendingSecretSaves.delete(run));
     return run;
   }

   // Route Apply (Task 7) and Verify (Task 12) await this, so they always act on
   // what main has stored rather than on a save still in flight.
   export async function flushSecretSaves(): Promise<void> {
     while (pendingSecretSaves.size > 0) {
       await Promise.allSettled([...pendingSecretSaves]);
     }
   }

   // Main broadcasts credentials-changed after every save, removal, undo and
   // reset; each window mirrors the per-field presence into its sentinels.
   export function subscribeCredentialChanges(): () => void {
     const credentials = isBrowser ? window.electronAPI?.credentials : undefined;
     if (!credentials?.onChanged) return () => {};
     return credentials.onChanged((change) => {
       const patch: Record<string, string> = {};
       const caches = new Set<Parameters<typeof invalidateApiKeyCaches>[0]>();
       for (const [storeKey, [id, field]] of Object.entries(SECRET_SLOTS) as [
         SecretStoreKey,
         readonly [CredentialSlotId, CredentialField],
       ][]) {
         const stored = change.fields[field];
         if (id !== change.id || stored === undefined) continue;
         patch[storeKey] = stored ? "__stored__" : "";
         caches.add(SECRET_CACHE[storeKey]);
       }
       if (Object.keys(patch).length === 0) return;
       useSettingsStore.setState(patch as Partial<SettingsState>);
       for (const provider of caches) invalidateApiKeyCaches(provider, { persistEnv: false });
     });
   }
