function recordingApi() {
  const calls = [];
  const subscriptions = {};
  const ok = (name, value) => async () => {
    calls.push(name);
    return value;
  };
  const forbidden = (name) => () => {
    calls.push(name);
    throw new Error(`${name} must not run while loading route facts`);
  };

    subscriptions[name] = cb;
    return () => {};

      list: ok("credentials.list", [{ id: "openai", present: true, generation: 1 }]),
      onChanged: subscribe("credentials"),
      save: forbidden("credentials.save"),
      remove: forbidden("credentials.remove"),
    },
    keyring: {
      list: ok("keyring.list", []),
      onChanged: subscribe("keyring"),
      verify: forbidden("keyring.verify"),

      codexModels: forbidden("personalInference.codexModels"),

    getTinfoilChatModels: forbidden("getTinfoilChatModels"),
    privacy: { getSealed: ok("privacy.getSealed", false), onSealedChange: subscribe("sealed") },

    chat = routes.useResolvedRoute("chat");
    return null;

  await React.act(async () => {
    root.render(React.createElement(Probe));

   // Deep-merges a per-locale fragment into every src/locales/<lang>/translation.json.
   // Usage: node scripts/merge-locale-keys.js <fragment.json> [localesRoot]
   // The fragment is { "en": {...}, "de": {...}, ... } and must cover every locale,
   // so a new key can never land in English only.

   const path = require("node:path");

   const [fragmentPath, rootArg] = process.argv.slice(2);
   if (!fragmentPath) {
     console.error("Usage: node scripts/merge-locale-keys.js <fragment.json> [localesRoot]");
     process.exit(2);
   }
   const root = rootArg ?? path.join(__dirname, "..", "src", "locales");
   const fragment = JSON.parse(fs.readFileSync(fragmentPath, "utf8"));
   const langs = fs
     .readdirSync(root)
     .filter((d) => fs.existsSync(path.join(root, d, "translation.json")));
   const missing = langs.filter((lang) => !fragment[lang]);
   if (missing.length) {
     console.error(`Missing locales in fragment: ${missing.join(", ")}`);
     process.exit(1);
   }

   const isObject = (v) => v !== null && typeof v === "object" && !Array.isArray(v);
   function merge(target, source) {
     for (const [key, value] of Object.entries(source)) {
       target[key] = isObject(value) ? merge(isObject(target[key]) ? target[key] : {}, value) : value;
     }
     return target;
   }

   for (const lang of langs) {
     const file = path.join(root, lang, "translation.json");
     const json = JSON.parse(fs.readFileSync(file, "utf8"));
     fs.writeFileSync(file, JSON.stringify(merge(json, fragment[lang]), null, 2) + "\n");
   }
   console.log(`Merged ${path.basename(fragmentPath)} into ${langs.length} locales.`);

     known: false,

     return useSealedStore.getState().sealed;
   }

   let subscribed = false;
   // Bumped by every seal event. The first fetch applies only if no event
   // arrived after it started, so a slow fetch can never undo a newer event.
   let eventVersion = 0;

   export function ensureSealedSubscribed(): void {
     if (subscribed || typeof window === "undefined") return;
     subscribed = true;
     const privacy = window.electronAPI?.privacy;
     if (!privacy?.getSealed && !privacy?.onSealedChange) {
       // This build has no sealed mode, so nothing can be sealed.
       useSealedStore.setState({ sealed: false, known: true });
       return;
     }
     privacy.onSealedChange?.((sealed) => {
       eventVersion += 1;
       useSealedStore.setState({ sealed: sealed === true, known: true });
     });
     const startedAt = eventVersion;
     Promise.resolve()
       .then(() => privacy.getSealed?.())
       .then(
         (sealed) => {
           if (typeof sealed !== "boolean" || eventVersion !== startedAt) return;
           useSealedStore.setState({ sealed, known: true });
         },
         () => {
           // Unknown stays unknown; the next event settles it.
         }
       );

   // Facts the route resolver needs that the settings store does not hold.
   // Every call here is local to this machine: model inventories on disk, main's
   // credential presence (Codex = a file stat in main) and keyring health. Never

     | "sherpa"
     | "llm"
     | "speakerId"

     credentials: null,
     rejected: new Set<CredentialId>(),
     speakerIdModelsReady: null,

     facts: EMPTY_ROUTE_FACTS,

   const requested: Partial<Record<RouteFactKey, number>> = {};
   const published: Partial<Record<RouteFactKey, number>> = {};

   async function track<T, V>(
     key: RouteFactKey,
     call: () => Promise<T> | undefined,
     read: (raw: T) => V | undefined,
     apply: (facts: RouteFacts, value: V) => RouteFacts
   ): Promise<void> {
     const generation = (requested[key] = (requested[key] ?? 0) + 1);
     let value: V | undefined;
     try {
       const pending = call();
       if (!pending) return; // this build has no such call; the fact stays as it is
       value = read(await pending);
     } catch {
       return; // transient failure: keep the last known value
     }
     if (value === undefined || generation <= (published[key] ?? 0)) return;
     published[key] = generation;
     useRouteFactsStore.setState((s) => ({ facts: apply(s.facts, value) }));
   }

   function downloadedIds(result: ModelListResult | undefined): Set<string> | undefined {
     if (!result || result.success === false || !Array.isArray(result.models)) return undefined;
     return new Set(result.models.filter((m) => m.downloaded).map((m) => m.model));
   }

   function withInstalled(
     facts: RouteFacts,
     family: "whisper" | "sherpa" | "llm",
     ids: ReadonlySet<string>
   ): RouteFacts {
     return { ...facts, installed: { ...facts.installed, [family]: ids } };
   }

   // Main's per-credential generation orders presence updates, so a list read
   // before a save can never undo the save's event.
   const credentialGenerations = new Map<CredentialId, number>();

   function mergeCredentials(facts: RouteFacts, rows: readonly CredentialRow[]): RouteFacts {
     const next = new Map(facts.credentials ?? []);
     let changed = facts.credentials === null;
     for (const row of rows) {
       if (!row || typeof row.id !== "string") continue;
       const generation = typeof row.generation === "number" ? row.generation : 0;
       if (generation < (credentialGenerations.get(row.id) ?? -1)) continue;
       credentialGenerations.set(row.id, generation);
       if (next.get(row.id) !== (row.present === true)) changed = true;
       next.set(row.id, row.present === true);
     }
     return changed ? { ...facts, credentials: next } : facts;
   }

     whisper: (api) =>
       track("whisper", () => api.listWhisperModels?.(), downloadedIds, (f, ids) =>
         withInstalled(f, "whisper", ids)

     sherpa: (api) =>

         withInstalled(f, "sherpa", ids)

         (f, ids) => withInstalled(f, "llm", ids)
       ),
     speakerId: (api) =>
       track(
         "speakerId",
         () => api.getDiarizationModelStatus?.(),
         (r) => (typeof r?.modelsDownloaded === "boolean" ? r.modelsDownloaded : undefined),
         (f, ready) => ({ ...f, speakerIdModelsReady: ready })
       ),
     search: (api) =>
       track(
         "search",
         () => api.getSearchModelStatus?.(),
         (r) => (typeof r?.downloaded === "boolean" ? r.downloaded : undefined),
         (f, ready) => ({ ...f, searchModelReady: ready })
       ),
     credentials: (api) =>
       track(
         "credentials",
         () => api.credentials?.list?.(),
         (rows) => (Array.isArray(rows) ? rows : undefined),
         mergeCredentials
       ),
     rejected: (api) =>
       track(
         "rejected",
         () => api.keyring?.list?.(),
         (rows) =>
           Array.isArray(rows)
             ? new Set(rows.filter((r) => r.status === "rejected").map((r) => r.id))
             : undefined,
         (f, rejected) => ({ ...f, rejected })
       ),
   };

   export async function loadRouteFacts(
     api: RouteFactsApi | undefined,
     keys: readonly RouteFactKey[] = ROUTE_FACT_KEYS
   ): Promise<void> {
     if (!api) return;
     await Promise.all(keys.map((key) => LOADERS[key](api)));
   }

   const currentApi = (): RouteFactsApi | undefined =>
     typeof window === "undefined" ? undefined : (window.electronAPI as unknown as RouteFactsApi);

   export function refreshRouteFacts(keys: readonly RouteFactKey[] = ROUTE_FACT_KEYS): Promise<void> {
     return loadRouteFacts(currentApi(), keys);
   }

   function applyCredentialChange(event: CredentialRow): void {
     if (useRouteFactsStore.getState().facts.credentials === null) {
       // Main has not listed presence yet: read the full list rather than
       // assume anything about the credentials this event does not name.
       void refreshRouteFacts(["credentials"]);
     } else {
       useRouteFactsStore.setState((s) => ({ facts: mergeCredentials(s.facts, [event]) }));
     }
     // Main resets a key's health on save and remove; re-read the verdict.
     void refreshRouteFacts(["rejected"]);
   }

   // Facts change on focus, after model downloads or deletes anywhere
   // (LOCAL_MODELS_CHANGED_EVENT, src/hooks/useModelDownload.ts:21; the cache
   // clear, src/components/SettingsPage.tsx:1478), and when main reports a
   // credential or keyring change.
   const REFRESH_EVENTS = ["focus", "openwhispr-models-cleared", "openwhispr-local-models-changed"];

   let subscribed = false;
   export function ensureRouteFactsSubscribed(): void {
     if (subscribed || typeof window === "undefined") return;
     subscribed = true;
     const api = currentApi();
     void refreshRouteFacts();
     api?.credentials?.onChanged?.(applyCredentialChange);
     api?.keyring?.onChanged?.(() => void refreshRouteFacts(["rejected"]));
     for (const event of REFRESH_EVENTS) {
       window.addEventListener(event, () => void refreshRouteFacts());
     }
   }

       window.removeEventListener("storage", bump);
     };
   }
   const readProviderSearchVersion = () => providerSearchVersion;

   export function useRouteCopy(): RouteCopy {
     const { t, i18n: instance } = useTranslation();
     const language = instance?.language || "en";
     return useMemo(() => routeCopy(t, language), [t, language]);
   }

   export function useResolvedRoutes(): ResolvedRoute[] {
     const settings = useSettingsStore();
     const facts = useRouteFactsStore((s) => s.facts);
     const sealed = useSealedStore((s) => s.sealed);
     const copy = useRouteCopy();
     const searchVersion = useSyncExternalStore(
       subscribeProviderSearch,
       readProviderSearchVersion,
       readProviderSearchVersion
     );
     useEffect(() => {
       ensureRouteFactsSubscribed();
       ensureSealedSubscribed();
     }, []);
     return useMemo(
       () => resolveRoutes(buildRouteSnapshot(settings, facts, copy, sealed)),
       // searchVersion is not read here: it re-runs the builder, which reads localStorage.
       // eslint-disable-next-line react-hooks/exhaustive-deps
       [settings, facts, copy, sealed, searchVersion]
     );
   }

   export function useResolvedRoute(id: RouteId): ResolvedRoute {
     const routes = useResolvedRoutes();
     return routes.find((route) => route.id === id) as ResolvedRoute;
   }
