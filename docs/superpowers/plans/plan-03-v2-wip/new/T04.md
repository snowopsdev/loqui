### Task 4: Settings paths, in-modal navigation and the support-models tab (SET-Q17)

Closes I13 for the support-model rows (the "Downloaded" status uses the locality role, never `text-success`, which measures 3.30:1 on white) and the Review Focus item "nothing flashes Download before facts load". It also delivers SET-Q17 and the navigation plumbing every later task uses.

Every `ResolvedRoute.settingsPath` (`"stt/dictation"`, `"llms/supportModels"`, …) must open Settings at that exact place, both from outside the modal (`ControlPanel`'s `onOpenSettings(section?: string)` feeds `SettingsModal`'s `initialSection`, `src/components/ControlPanel.tsx:669`) and from inside it (Patchbay rows in Task 10, search results in Task 16). SET-Q17 moves speaker identification and semantic search out of Privacy & Data into a Language Models sub-tab, with their sizes. Once a support model is downloaded, the route facts are refreshed (Task 3), so the `speakerId` and `semanticSearch` chips leave "Download" without waiting for a focus event.

   ```ts
   import type { SettingsSectionType } from "../components/SettingsPage";

   // Settings locations are "<section>" or "<section>/<subTab>". Route
   // settingsPaths (src/lib/routeGraph.ts) and every legacy deep link resolve
   // here, and anything unknown falls back to Preferences rather than a blank pane.

   export interface SettingsLocation {
     section: SettingsSectionType;
     subTab?: string;
   }

   export const SETTINGS_SECTIONS: readonly SettingsSectionType[] = [
     "general",
     "hotkeys",
     "speechToText",
     "llms",
     "privacyData",
     "system",
   ];

   // Sub-tab ids, in tab order. SettingsPage builds its tab unions from these so a
   // route path can never name a tab that does not exist.
   export const SPEECH_SUB_TABS = ["dictation", "noteRecording", "upload"] as const;
   export const LLM_SUB_TABS = [
     "dictationCleanup",
     "dictationAgent",
     "dictationTranslation",
     "noteFormatting",
     "chatIntelligence",
     "supportModels",
   ] as const;

   const SUB_TABS: Partial<Record<SettingsSectionType, readonly string[]>> = {
     speechToText: SPEECH_SUB_TABS,
     llms: LLM_SUB_TABS,
   };

   // The old AI Models sidebar had four items (transcription, meetings,
   // intelligence, agentMode) — they now collapse into two: speechToText + llms.
   // Legacy deep-links land on the matching sub-tab via LEGACY_SUB_TAB.
   // "dictationAgent" is a live deep-link (the Home GPU banner), not a legacy alias.
   // "stt" is the section prefix route paths use ("stt/dictation").
   export const SECTION_ALIASES: Record<string, SettingsSectionType> = {
     aiModels: "llms",
     agentConfig: "llms",
     agentMode: "llms",
     dictationAgent: "llms",
     intelligence: "llms",
     meetings: "llms",
     prompts: "llms",
     transcription: "speechToText",
     uploadTranscription: "speechToText",
     stt: "speechToText",
     softwareUpdates: "system",
     privacy: "privacyData",
     permissions: "privacyData",
     developer: "system",
   };

   export const LEGACY_SUB_TAB: Record<string, string> = {
     transcription: "dictation",
     uploadTranscription: "upload",
     dictationAgent: "dictationAgent",
     meetings: "noteFormatting",
     intelligence: "dictationCleanup",
     agentMode: "chatIntelligence",
     agentConfig: "chatIntelligence",
     aiModels: "dictationCleanup",
     prompts: "dictationCleanup",
   };

   export function parseSettingsPath(path: string | null | undefined): SettingsLocation {
     if (!path) return { section: "general" };
     const [head, sub] = path.split("/", 2);
     const section = (SETTINGS_SECTIONS as readonly string[]).includes(head)
       ? (head as SettingsSectionType)
       : SECTION_ALIASES[head];
     if (!section) return { section: "general" };
     const subTab = sub || LEGACY_SUB_TAB[head];
     const known = SUB_TABS[section];
     return subTab && (!known || known.includes(subTab)) ? { section, subTab } : { section };
   ```

4. Add `formatDecimalBytes` to `src/utils/formatBytes.ts`. Model hosts publish decimal sizes ("34.8 MB", "91 MB"), and the existing `formatBytes` is 1024-based:

   ```ts
   // Decimal (1000-based) units, matching the sizes model hosts publish.
   export function formatDecimalBytes(bytes: number, decimals = 1): string {
     if (!Number.isFinite(bytes) || bytes <= 0) return "0 MB";
     const units = ["B", "KB", "MB", "GB", "TB"];
     const i = Math.min(units.length - 1, Math.floor(Math.log10(bytes) / 3));
     return `${parseFloat((bytes / 1000 ** i).toFixed(decimals))} ${units[i]}`;
   }
   ```

5. Replace the whole of `src/components/settings/LocalSupportModels.tsx`. The status reads and `download()` keep their behaviour; what changes is that readiness starts unknown (no Download flash), sizes and the device ring show, "Downloaded" uses the locality role, and a finished download refreshes route facts.

   ```tsx
   import { useEffect, useState } from "react";
   import { useTranslation } from "react-i18next";
   import modelRegistryData from "../../models/modelRegistryData.json";
   import { refreshRouteFacts } from "../../stores/routeFactsStore";
   import { formatDecimalBytes } from "../../utils/formatBytes";
   import { Button } from "../ui/button";
   import { LocalityRing } from "../ui/LocalityRing";
   import { SectionHeader, SettingsPanel, SettingsPanelRow, SettingsRow } from "../ui/SettingsSection";

   type ModelKind = "meeting" | "search";

   // Sizes shown before download. Diarization is the sum of the registry entries;
   // the embedding model (all-MiniLM-L6-v2 ONNX plus tokenizer) is fetched by
   // scripts/download-minilm.js and is about 91 MB.
   const SUPPORT_MODEL_BYTES: Record<ModelKind, number> = {
     meeting: Object.values(modelRegistryData.diarizationModels).reduce(
       (sum, model) => sum + model.sizeMb * 1_000_000,
       0
     ),
     search: 91_000_000,
   };

   export default function LocalSupportModels() {
     const { t } = useTranslation();
     // null = not read yet. Nothing says "Download" until main has answered.
     const [ready, setReady] = useState<Record<ModelKind, boolean | null>>({
       meeting: null,
       search: null,
     });
     const [checking, setChecking] = useState(true);
     const [busy, setBusy] = useState<ModelKind | null>(null);
     const [error, setError] = useState("");

     useEffect(() => {
       let mounted = true;
       const api = window.electronAPI;
       void Promise.all([api.getDiarizationModelStatus?.(), api.getSearchModelStatus?.()])
         .then(([meeting, search]) => {
           if (mounted)
             setReady({ meeting: !!meeting?.modelsDownloaded, search: !!search?.downloaded });
         })
         .catch((error) => {
           if (mounted) setError(String(error));
         })
         .finally(() => {
           if (mounted) setChecking(false);
         });
       return () => {
         mounted = false;
       };
     }, []);

     const download = async (kind: ModelKind) => {
       setBusy(kind);
       setError("");
       try {
         const result =
           kind === "meeting"
             ? await window.electronAPI.downloadDiarizationModels?.()
             : await window.electronAPI.downloadSearchModel?.();
         if (!result?.success) throw new Error(result?.error || t("personal.supportDownloadFailed"));
         setReady((previous) => ({ ...previous, [kind]: true }));
         // The route graph reads these two as route facts (speakerId, semanticSearch).
         void refreshRouteFacts();
       } catch (error) {
         setError(String(error));
       } finally {
         setBusy(null);
       }
     };

     return (
       <div className="space-y-3">
         <SectionHeader
           title={t("settingsPage.llms.supportModels.title")}
           description={t("settingsPage.llms.supportModels.description")}
         />
         <SettingsPanel>
           {(["meeting", "search"] as const).map((kind) => (
             <SettingsPanelRow key={kind}>
               <SettingsRow
                 label={t(`personal.supportModels.${kind}`)}
                 description={t(`personal.supportModels.${kind}Description`)}
               >
                 <div className="flex items-center gap-2">
                   <LocalityRing locality="device" label={t("locality.device")} size={18} />
                   {checking ? (
                     <span className="text-xs font-medium text-muted-foreground">
                       {t("settingsPage.llms.supportModels.checking")}
                     </span>
                   ) : ready[kind] ? (
                     <span className="text-xs font-medium text-locality-device">
                       {t("settingsPage.llms.supportModels.downloaded")}
                     </span>
                   ) : (
                     <Button
                       size="sm"
                       variant="outline"
                       disabled={!!busy}
                       onClick={() => void download(kind)}
                     >
                       {busy === kind
                         ? t("personal.downloading")
                         : t("settingsPage.llms.supportModels.download", {
                             size: formatDecimalBytes(SUPPORT_MODEL_BYTES[kind]),
                           })}
                     </Button>
                   )}
                 </div>
               </SettingsRow>
             </SettingsPanelRow>
           ))}
         </SettingsPanel>
         {error && (
           <p role="alert" className="text-xs text-destructive">
             {error}
           </p>
         )}
       </div>
     );
   }
   ```
