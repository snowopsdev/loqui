### Task 6: Route planner: one patch for preview, commit and Undo

Closes **I5** (preview, commit and Undo describe different configurations) and **I6** (a one-route change silently changes inheriting routes). It also provides the classification (`kind`) and click-time blocker that Task 7 uses to close I1.

There is one computation, and nothing else may write a route:
- **Shared provider switches.** `src/lib/providerSwitchPlans.ts` holds the provider-switch logic, extracted from the store's `switchCloudTranscriptionProvider` and `switchReasoningProvider`, including the per-provider model memory. Both store actions now call these planners and then `applyRoutePatch`, so the store and the planner run the same implementation.
- **The planner.** `src/lib/routePlanner.ts` turns a target into the exact raw store writes (the patch). It simulates the patch on a copy of the state and resolves all 11 routes before and after. It lists every route whose destination, model, payload or state changes, including coupled ones: Note recording or Upload inheriting dictation, and assistant vision following the assistant. It also records the prior raw value of every written key (the Undo patch).
- **One patch for everything.** Preview is the after-resolution of that patch. Commit is `applyRoutePatch(patch)`. Undo is `applyRoutePatch(undoPatch)`. They cannot disagree.

   ```ts
   export function resolvedScopeField(

   // Every provider shares one model slot per scope, so a plain provider write
   // destroys the outgoing provider's model (the "Parasail" wipe). This plan
   // remembers the outgoing model and restores the incoming provider's last one.
   export function planTranscriptionProviderSwitch(
     state: SettingsState,
     context: TranscriptionPolicyContext,
     providerId: string
   ): SettingsPatch {
     const patch: SettingsPatch = {};
     const keys = TRANSCRIPTION_CONTEXT_KEYS.find((entry) => entry.context === context);
     if (!keys) return patch;
     // Meeting/upload raw keys default to "" (inherit from dictation), so work
     // against resolved values. Restoring writes a concrete model — the scope
     // stops inheriting the dictation model from that point on.
     const outgoingProvider = (state[keys.provider] as string) || state.cloudTranscriptionProvider;
     if (outgoingProvider === providerId) {
       write(patch, keys.provider, providerId);
       return patch;
     }
     const outgoingModel = (state[keys.model] as string) || state.cloudTranscriptionModel;
     const memory = { ...state.transcriptionModelByProvider };
     if (outgoingProvider && outgoingModel) {
       memory[`${context}:${outgoingProvider}`] = outgoingModel;
       patch.transcriptionModelByProvider = memory;
     }
     const remembered = memory[`${context}:${providerId}`];
     write(patch, keys.provider, providerId);
     write(
       patch,
       keys.model,
       remembered && transcriptionModelBelongsToProvider(providerId, remembered, context)
         ? remembered
         : defaultTranscriptionModel(providerId, context)
     );
     return patch;
   }

   // LLM twin. The caller supplies the provider's default model because dynamic
   // catalogs (Tinfoil) live in the UI.
   export function planReasoningProviderSwitch(
     state: SettingsState,
     scope: InferenceScope,
     providerId: string,
     fallbackModel = ""
   ): SettingsPatch {
     const keys = INFERENCE_SCOPES[scope].storeKeys;
     const currentProvider = resolvedScopeField(state, scope, "provider");
     const currentModel = resolvedScopeField(state, scope, "model");
     const memory = { ...state.reasoningModelByProvider };
     const patch: SettingsPatch = {};
     // Only remember a model the outgoing provider actually owns — after a
     // local/cloud mode round-trip the shared slot can hold a foreign id.
     if (currentProvider && currentProvider !== providerId) {
       if (reasoningModelBelongsToProvider(currentProvider, currentModel)) {
         memory[`${scope}:${currentProvider}`] = currentModel;
         patch.reasoningModelByProvider = memory;
       }
     } else if (reasoningModelBelongsToProvider(providerId, currentModel)) {
       // Reselecting the current provider with a model it owns: keep it.
       write(patch, keys.provider, providerId);
       return patch;
     }
     const remembered = memory[`${scope}:${providerId}`];
     write(patch, keys.provider, providerId);
     write(
       patch,
       keys.model,
   ```

```ts
const merged = (state: SettingsState, patch: Patch) => ({ ...state, ...patch }) as SettingsState;

function sttTargetPatch(state: SettingsState, context: SttContext, target: RouteTarget): Patch {
  const k = STT_KEYS[context];
  const patch: Patch = {
    [k.mode]: target.mode,
    [k.useLocal]: target.mode === "local",
    [k.cloudMode]: "byok",
  };
  if (target.mode === "local") {
    if (target.provider !== undefined) patch[k.localProvider] = target.provider;
    if (target.model !== undefined) {
      const provider = (target.provider ?? state[k.localProvider]) as LocalTranscriptionProvider;
      patch[k.localModel[provider] ?? k.localModel.whisper] = target.model;
    }
    return patch;
  }
  if (target.mode === "providers") {
    if (target.provider) {
      Object.assign(patch, planTranscriptionProviderSwitch(merged(state, patch), context, target.provider));
    } else {
      patch[k.cloudProvider] = "";
    }
    if (target.model !== undefined) patch[k.cloudModel] = target.model;
    if (target.baseUrl !== undefined) patch[k.cloudBaseUrl] = target.baseUrl;
    return patch;
  }
  if (k.remoteUrl) patch[k.remoteUrl] = target.remoteUrl;
  if (k.remoteModel && target.model !== undefined) patch[k.remoteModel] = target.model;
  return patch;
}

function providerPatch(
  state: SettingsState,
  scope: LlmRouteScope,
  target: ProvidersTarget,
  options: { vision: boolean; mode: InferenceMode }
): Patch {
  const keys = INFERENCE_SCOPES[scope].storeKeys as Record<string, string>;
  let patch: Patch;
  if (target.provider) {
    patch = {
      ...planReasoningProviderSwitch(
        state,
        scope,
        target.provider,
        defaultCloudModel(target.provider, { vision: options.vision })
      ),
    };
  } else {
    // "Choose a provider". Like InferenceConfigEditor.tsx:109-112, a model
    // whose provider the new mode cannot use goes with it.
    patch = { [keys.provider]: "" };
    if (!isProviderValidForMode(resolvedScopeField(state, scope, "provider"), options.mode)) {
      patch[keys.model] = "";
    }
  }
  if (target.model !== undefined) patch[keys.model] = target.model;
  if (target.baseUrl !== undefined && keys.cloudBaseUrl) patch[keys.cloudBaseUrl] = target.baseUrl;
  return patch;
}

function llmTargetPatch(state: SettingsState, scope: LlmRouteScope, target: RouteTarget): Patch {
  const keys = INFERENCE_SCOPES[scope].storeKeys as Record<string, string>;

  if (scope === "dictationAgentVision") {
    if (target.mode !== "providers") {
      throw new Error("Assistant vision runs on a provider: use a providers target");
    }
    // Only the override flag and the vision provider, model and endpoint. Never
    // screen context, never the vision mode (settingsStore.ts:1757-1758).
    const patch: Patch = {
      useDictationAgentVisionModel: true,
      ...providerPatch(state, scope, target, { vision: true, mode: "providers" }),
    };
    // dictationAgentInference.js:72 treats the override as chosen only when
    // its own model is set; a switch that keeps the inherited model must write it.
    const after = merged(state, patch);
    if (!after.dictationAgentVisionModel) {
      patch.dictationAgentVisionModel =
        resolvedScopeField(after, scope, "model") ||
        defaultCloudModel(target.provider, { vision: true });
    }
    return patch;
  }

  const currentMode = state[keys.mode as keyof SettingsState] as InferenceMode;

  if (target.mode === "local") {
    const patch: Patch = { [keys.mode]: "local", [keys.cloudMode]: "byok" };
    if (target.provider !== undefined) {
      const current = resolvedScopeField(state, scope, "provider");
      patch[keys.provider] = target.provider;
      patch[keys.model] =
        target.model ?? (target.provider === current ? resolvedScopeField(state, scope, "model") : "");
      return patch;
    }
    // InferenceConfigEditor.tsx:109-112: a provider the new mode cannot use is cleared with its model.
    if (!isProviderValidForMode(resolvedScopeField(state, scope, "provider"), "local")) {
      patch[keys.provider] = "";
      patch[keys.model] = "";
    }
    if (target.model !== undefined) patch[keys.model] = target.model;
    return patch;
  }

  if (target.mode === "self-hosted") {
    const patch: Patch = {
      [keys.mode]: "self-hosted",
      [keys.cloudMode]: "byok",
      [keys.remoteUrl]: target.remoteUrl,
    };
    // A cloud or local model id means nothing to your server.
    if (target.model !== undefined) patch[keys.model] = target.model;
    else if (currentMode !== "self-hosted") patch[keys.model] = "";
    return patch;
  }

  const mode: InferenceMode = ENTERPRISE_PROVIDER_IDS.has(target.provider) ? "enterprise" : "providers";
  const patch: Patch = { [keys.mode]: mode, [keys.cloudMode]: "byok" };
  return Object.assign(patch, providerPatch(merged(state, patch), scope, target, { vision: false, mode }));
}

function sameValue(a: unknown, b: unknown): boolean {
  if ((a && typeof a === "object") || (b && typeof b === "object")) {
    return JSON.stringify(a ?? null) === JSON.stringify(b ?? null);
  }
  return a === b;
}

function fingerprint(route: ResolvedRoute): string {
  return JSON.stringify([route.locality, route.providerId, route.host, route.model, route.sends, route.state]);
}

// Instant when nothing new leaves the device: the route ends on the device,
// or it keeps the same recipient (locality, provider and host) and its
// payload did not grow. An off-device endpoint without a known host is never
// assumed to be the same one.
function addsNoRecipientOrPayload(
  change: RouteChange,
  beforeSnapshot: RouteSettingsSnapshot,
  afterSnapshot: RouteSettingsSnapshot
): boolean {
  const { before, after } = change;
  if (after.locality === "device") return true;
  if (
    after.locality !== before.locality ||
    after.providerId !== before.providerId ||
    after.host !== before.host
  ) {
    return false;
  }
  if (after.host === null && (after.providerId === null || after.providerId === "custom")) return false;
  const had = new Set<string>(routePayloadParts(change.id, beforeSnapshot));
  return routePayloadParts(change.id, afterSnapshot).every((part) => had.has(part));
}

const BLOCKER_PRIORITY: ApplyBlocker[] = [
  "sealed",
  "unknown",
  "chooseProvider",
  "needsKey",
  "needsUrl",
  "unsupported",
  "needsModel",
];

function routeBlocker(route: ResolvedRoute): ApplyBlocker | null {
  if (route.locality === "device") return null;
  const state = route.state;
  if (state.status === "unknown") return "unknown";
  if (state.status === "needsUrl") return "needsUrl";
  if (state.status === "unreachable") return state.detail === "sealed" ? "sealed" : "unsupported";
  if (state.status === "needsKey") {
    if (state.provider === "") return "chooseProvider";
    return route.providerId && SETUP_AFTER_APPLY.has(route.providerId) ? null : "needsKey";
  }
  // A remote text route picks its model in the provider editor after switching.
  if (state.status === "needsModel") return state.sizeBytes !== undefined ? "needsModel" : null;
  return null;
}

export function planPatch(
  state: SettingsState,
  ctx: PlanContext,
  routeId: RouteId,
  patch: SettingsPatch
): RoutePlan {
  const written: Patch = {};
  const undo: Patch = {};
  const current = state as unknown as Patch;
  for (const [key, value] of Object.entries(patch)) {
    if (value === undefined) continue;
    if (!ROUTE_PATCH_KEY_SET.has(key)) throw new Error(`Route patches cannot write ${key}`);
    if (sameValue(current[key], value)) continue;
    written[key] = value;
    undo[key] = current[key];
  }
  if (Object.keys(written).length === 0) {
    return { routeId, patch: {}, undoPatch: {}, changes: [], kind: "noop", blocker: null };
  }

  const beforeSnapshot = buildRouteSnapshot(state, ctx.facts, ctx.copy, ctx.sealed);
  const afterSnapshot = buildRouteSnapshot(merged(state, written), ctx.facts, ctx.copy, ctx.sealed);
  const afterById = new Map(resolveRoutes(afterSnapshot).map((route) => [route.id, route]));
  const changes: RouteChange[] = [];
  for (const before of resolveRoutes(beforeSnapshot)) {
    const after = afterById.get(before.id);
    if (after && fingerprint(after) !== fingerprint(before)) {
      changes.push({ id: before.id, before, after, coupled: before.id !== routeId });
    }
  }
  changes.sort((a, b) => Number(a.coupled) - Number(b.coupled));

  const instant = changes.every((change) =>
    addsNoRecipientOrPayload(change, beforeSnapshot, afterSnapshot)
  );
  const found = new Set(changes.map((change) => routeBlocker(change.after)));
  const blocker = BLOCKER_PRIORITY.find((candidate) => found.has(candidate)) ?? null;

  return {
    routeId,
    patch: written as SettingsPatch,
    undoPatch: undo as SettingsPatch,
    changes,
    kind: instant ? "instant" : "staged",
    blocker,
  };
}

export function planRouteChange(
  state: SettingsState,
  ctx: PlanContext,
  id: RouteId,
  target: RouteTarget
): RoutePlan {
  const context = STT_CONTEXT_OF[id];
  const scope = LLM_SCOPE_OF[id];
  if (!context && !scope) {
    throw new Error(`${id} always runs on this device; it has no destination to change`);
  }
  const patch = context ? sttTargetPatch(state, context, target) : llmTargetPatch(state, scope, target);
  return planPatch(state, ctx, id, patch as SettingsPatch);
}
```

Notes for the implementer:
- `routeBlocker` returns null for `off` routes because their state is `off`, so an off route never blocks (INTERFACES: "every non-off changed route").
- The `state.status === …` comparisons narrow explicitly, because `src/tsconfig.json` is not strict.
- `planPatch` drops keys whose value already matches. Re-selecting where a route already runs is therefore `noop`, and `undoPatch` holds exactly the keys that change.
- `undoPatch` holds the raw prior value. For inherited meeting and upload fields, that is `""` (inheritance restored). For the vision flags, it is the prior `useDictationAgentVisionModel` and the prior vision provider and model.
