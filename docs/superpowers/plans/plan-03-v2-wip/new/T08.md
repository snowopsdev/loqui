// Mounted behaviour harness in the personalWorkspace pattern
// (test/components/personalWorkspace.test.js:18-41): every probe calls a
// component as a plain function and returns null, so hooks and effects run
// but no host element is ever committed to installHookDom's fake DOM.
const React = require("react");
const { createRoot } = require("react-dom/client");
const { installHookDom } = require("./rendererTestHarness");

function createProbeRoot(t, { wrap } = {}) {
  const container = installHookDom(t);
  const root = createRoot(container);
  const slots = new Map();
  t.after(async () => {
    await React.act(async () => root.unmount());
  });

  function Probe({ slot }) {
    slot.tree = slot.render();
    return null;
  }

  async function draw() {
    const probes = [...slots.entries()].map(([name, slot]) =>
      React.createElement(Probe, { key: `${name}:${slot.generation}`, slot })
    );
    const tree = React.createElement(React.Fragment, null, ...probes);
    await React.act(async () => root.render(wrap ? wrap(tree) : tree));
  }

  return {
    // Mounting a name again remounts it (fresh hooks).
    async mount(name, render) {
      const generation = (slots.get(name)?.generation ?? 0) + 1;
      slots.set(name, { render, tree: null, generation });
      await draw();
      return () => slots.get(name).tree;
    },
    async act(fn) {
      await React.act(async () => {
        await fn();
      });
    },
    // Lets queued promises (flushSecretSaves, requestRouteChange) finish.
    async settle() {
      await React.act(async () => {
        await new Promise((resolve) => setTimeout(resolve, 0));
      });
    },
  };
}

function findAll(node, predicate, found = []) {
  if (Array.isArray(node)) {
    for (const item of node) findAll(item, predicate, found);
    return found;
  }
  if (!node || typeof node !== "object") return found;
  if (predicate(node)) found.push(node);
  findAll(node.props?.children, predicate, found);
  return found;
}

function findElement(node, predicate) {
  return findAll(node, predicate)[0] ?? null;
}

module.exports = { createProbeRoot, findAll, findElement };

         const result = await requestRouteChange(id, target, copy);
         if (result !== "staged") announce(result);
         return result;
       },
       [copy, announce]

#### Enable switches through the planner (INTEGRATION-A)

The page switches that turn a route on (cleanup, translation, assistant, screen context, the vision override) and the provider web-search toggle wrote the store or `localStorage` directly, so turning on a route that already pointed at a cloud provider, or switching web search on, sent new data with no review. They now commit through `useRouteChange().requestPatch`, which is Task 7's `requestRoutePatch`. Off is always instant, because an off route sends nothing. On is instant when the route resolves to this device and staged when it leaves the device. The vision override is planned like any recipient change. Each switch shows the live value, so a staged "on" stays off until Apply, and the Apply bar lists the change.

   ```tsx
   import type { RouteSwitchKey, SettingsPatch } from "../../../lib/providerSwitchPlans";
   import type { RouteId } from "../../../lib/routes";
   import { useSettingsStore } from "../../../stores/settingsStore";
   import { Toggle } from "../../ui/toggle";
   import { useRouteChange } from "./useRouteChange";

   export interface RouteSwitchProps {
     routeId: RouteId;
     settingKey: RouteSwitchKey;
     disabled?: boolean;
     /** Runs after the change is committed, never while it waits for Apply. */
     onApplied?: (enabled: boolean) => void;
   }

   // A switch that turns a route on or off. The planner decides (Task 6):
   // off, or on for a route that stays on this device, applies at once with
   // Undo; on for a route that leaves the device waits in the Apply bar. The
   // switch shows the live value, so a staged "on" stays off until Apply.
   export function RouteSwitch({ routeId, settingKey, disabled, onApplied }: RouteSwitchProps) {
     const checked = useSettingsStore((s) => s[settingKey]);
     const { requestPatch } = useRouteChange();
     return (
       <Toggle
         checked={checked}
         disabled={disabled}
         onChange={async (next) => {
           const outcome = await requestPatch(routeId, { [settingKey]: next } as SettingsPatch);
           if (outcome !== "staged" && outcome.applied) onApplied?.(next);
         }}
       />
     );
   }
   ```
