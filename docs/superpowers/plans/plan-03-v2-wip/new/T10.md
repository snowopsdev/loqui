   ```ts
     type ResolvedRoute,

   export type DestinationKind = "device" | "server" | "addServer" | "custom" | "provider";
   export type TaskKind = "speech" | "text";
   export type KeyState = "saved" | "missing" | "rejected" | "unknown" | "none";

   export interface RouteDestination {
     id: string;
     kind: DestinationKind;
     locality: Locality;
     providerId: string | null;
     host: string | null;
     urls: { speech: string | null; text: string | null };
   }

   export type Compatibility =
     | { ok: true }
     | {
         ok: false;
         reasonKey: string;
         params?: { provider?: string; server?: string; kind?: TaskKind };
       };

   export interface CompatContext {
     sealed: boolean;
     providersFor: (id: RouteId) => readonly string[];
     dictationServerHost: string | null;

   export const ADD_SERVER_ID = "network:new";

   const NO_URLS = { speech: null, text: null } as const;

   function hostOf(url: string | null | undefined): string | null {
     if (!url || !url.trim()) return null;
     try {
       return new URL(url.trim()).host || null;
     } catch {
       return null;
     }
   }

   export function taskKind(routeId: RouteId): TaskKind {
     return ROUTE_KIND[routeId] === "speech" ? "speech" : "text";
   }

   // Identity is a function of the resolved route alone. A retired service and
   // a server route with no usable address have no destination: their chip
   // already says why, and a card for them would invent a recipient.
   export function destinationKeyOf(
     route: Pick<ResolvedRoute, "locality" | "providerId" | "host" | "state">
   ): string | null {
     if (route.state.status === "unreachable" && route.state.detail === "retired") return null;
     if (route.locality === "device") return "device";
     if (route.locality === "network") return route.host ? `network:${route.host}` : null;
     if (!route.providerId) return null;
     if (route.providerId === "custom") return route.host ? `custom:${route.host}` : null;
     return route.providerId;
   }

   // Corti has two credential slots: speech uses the client pair, text uses the
   // API key (Task 1). Every other provider is its own slot; "custom" has none.
   export function credentialIdFor(
     providerId: string | null,
     kind: ResolvedRoute["kind"] | TaskKind
   ): CredentialId | null {
     if (!providerId || providerId === "custom") return null;
     if (providerId === "corti") return kind === "speech" ? "corti:speech" : "corti:text";
     return providerId as CredentialId;
   }

   function providerOfCredential(id: string): string {
     return id.startsWith("corti:") ? "corti" : id;
   }

   // The first URL of this kind that points at this host. Scan order is fixed
   // (dictation, upload, meeting; then the LLM scopes in declaration order), so
   // the result never depends on object iteration luck.
   function knownUrl(
     snapshot: RouteSettingsSnapshot,
     host: string,
     kind: TaskKind,
     field: "remoteUrl" | "cloudBaseUrl"
   ): string | null {
     const inputs: Array<{ remoteUrl?: string; cloudBaseUrl?: string }> =
       kind === "speech"
         ? [snapshot.stt.dictation, snapshot.stt.upload, snapshot.stt.meeting]
         : Object.values(snapshot.llm);
     for (const input of inputs) {
       const url = (input[field] ?? "").trim();
       if (url && hostOf(url) === host) return url;
     }
     return null;
   }

   export function buildDestinations(
     snapshot: RouteSettingsSnapshot,
     routes: readonly ResolvedRoute[]
   ): RouteDestination[] {
     const serverHosts = new Set<string>();
     const customHosts = new Set<string>();
     const providerIds = new Set<string>();
     // Off routes keep the destination they would use, but they send nothing, so
     // they do not create cards.
     for (const route of routes) {
       if (route.state.status === "off") continue;
       if (destinationKeyOf(route) === null) continue;
       if (route.locality === "network") serverHosts.add(route.host as string);
       else if (route.locality === "remote" && route.providerId === "custom") {
         customHosts.add(route.host as string);
       } else if (route.locality === "remote") providerIds.add(route.providerId as string);
     }
     for (const [id, present] of snapshot.facts.credentials ?? []) {
       if (present) providerIds.add(providerOfCredential(id));
     }

     const hosts = snapshot.capabilities.providerHosts;
     const device: RouteDestination = {
       id: "device", kind: "device", locality: "device", providerId: null, host: null, urls: { ...NO_URLS },
     };
     const servers = [...serverHosts].sort().map<RouteDestination>((host) => ({
       id: `network:${host}`,
       kind: "server",
       locality: "network",
       providerId: null,
       host,
       urls: {
         speech: knownUrl(snapshot, host, "speech", "remoteUrl"),
         text: knownUrl(snapshot, host, "text", "remoteUrl"),
       },
     }));
     const addServer: RouteDestination = {
       id: ADD_SERVER_ID, kind: "addServer", locality: "network", providerId: null, host: null, urls: { ...NO_URLS },
     };
     const customs = [...customHosts].sort().map<RouteDestination>((host) => ({
       id: `custom:${host}`,
       kind: "custom",
       locality: "remote",
       providerId: "custom",
       host,
       urls: {
         speech: knownUrl(snapshot, host, "speech", "cloudBaseUrl"),
         text: knownUrl(snapshot, host, "text", "cloudBaseUrl"),
       },
     }));
     const providers = [...providerIds]
       .sort((a, b) => providerDisplayName(a).localeCompare(providerDisplayName(b)) || a.localeCompare(b))
       .map<RouteDestination>((id) => ({
         id,
         kind: "provider",
         locality: "remote",
         providerId: id,
         host: Object.hasOwn(hosts, id) ? hosts[id] : null,
         urls: { ...NO_URLS },
       }));
     return [device, ...servers, addServer, ...customs, ...providers];
   }

   export function compatContext(
     snapshot: RouteSettingsSnapshot,
     sealed: boolean,
     providersFor: CompatContext["providersFor"]
   ): CompatContext {
     return { sealed, providersFor, dictationServerHost: hostOf(snapshot.stt.dictation.remoteUrl) };
   }

   export function checkCompatibility(
     routeId: RouteId,
     dest: RouteDestination,
     ctx: CompatContext
   ): Compatibility {
     if (routeId === "speakerId" || routeId === "semanticSearch") {
       return { ok: false, reasonKey: `patchbay.locked.${routeId}` };
     }
     if (ctx.sealed && dest.kind !== "device") return { ok: false, reasonKey: "patchbay.no.sealed" };
     // Screenshots go to a cloud vision model; the vision scope has no server URL of its own.
     if (routeId === "assistantVision" && dest.locality !== "remote") {
       return { ok: false, reasonKey: "patchbay.no.visionCloudOnly" };
     }
     if (dest.kind === "device") return { ok: true };
     const kind = taskKind(routeId);
     if (dest.kind === "addServer" || dest.kind === "server") {
       // Note recording's self-hosted option is not available (MeetingSettings.tsx:89).
       if (routeId === "noteRecordingStt") return { ok: false, reasonKey: "patchbay.no.meetingServer" };
       // Upload reads dictation's server and never writes it, so it can only go
       // to the server dictation already points at.
       if (routeId === "uploadStt") {
         return dest.kind === "server" && dest.host !== null && dest.host === ctx.dictationServerHost
           ? { ok: true }
           : { ok: false, reasonKey: "patchbay.no.uploadServer" };
       }
       if (dest.kind === "addServer" || dest.urls[kind] !== null) return { ok: true };
       return { ok: false, reasonKey: "patchbay.no.serverKind", params: { server: dest.host ?? "", kind } };
     }
     if (dest.providerId === null || !ctx.providersFor(routeId).includes(dest.providerId)) {
       return { ok: false, reasonKey: "patchbay.no.capability", params: { provider: dest.providerId ?? "" } };
     }
     if (dest.kind === "custom" && dest.urls[kind] === null) {
       return { ok: false, reasonKey: "patchbay.no.serverKind", params: { server: dest.host ?? "", kind } };
     }
     return { ok: true };
   }

   export function compatReason(compat: Extract<Compatibility, { ok: false }>, t: Translate): string {
     const p = compat.params ?? {};
     const options = {
       ...(p.provider !== undefined ? { provider: providerDisplayLabel(p.provider, t) } : {}),
       ...(p.server !== undefined ? { server: p.server } : {}),
       ...(p.kind !== undefined ? { kind: t(`patchbay.kind.${p.kind}`) } : {}),
     };
     return Object.keys(options).length > 0 ? t(compat.reasonKey, options) : t(compat.reasonKey);
   }

   // The target is built from the destination alone. "Add a server" starts with
   // an empty address, which the planner reports as needsUrl and Task 8's staged
   // panel asks for; no other task's URL is ever reused.
   export function targetForDestination(routeId: RouteId, dest: RouteDestination): RouteTarget {
     const kind = taskKind(routeId);
     switch (dest.kind) {
       case "device":
         return { mode: "local" };
       case "addServer":
         return { mode: "self-hosted", remoteUrl: "" };
       case "server":
         return { mode: "self-hosted", remoteUrl: dest.urls[kind] ?? "" };
       case "custom":
         return { mode: "providers", provider: "custom", baseUrl: dest.urls[kind] ?? "" };
       default:
         return { mode: "providers", provider: dest.providerId as string };
     }
   }

   // Key readiness for a provider card, from main-confirmed presence only
   // (rule 5). null kind (no task selected) is treated as text; callers do not
   // show a key line without a selection.
   export function destinationKeyState(
     dest: RouteDestination,
     facts: RouteFacts,
     kind: TaskKind | null
   ): KeyState {
     if (dest.kind !== "provider" || dest.providerId === null) return "none";
     if (facts.credentials === null) return "unknown";
     const id = credentialIdFor(dest.providerId, kind ?? "text");
     if (id === null || !facts.credentials.has(id)) return "none";
     if (facts.rejected.has(id)) return "rejected";
     return facts.credentials.get(id) === true ? "saved" : "missing";
   }

   export function destinationLabel(dest: RouteDestination, t: Translate): string {
     switch (dest.kind) {
       case "device":
         return t("locality.device");
       case "addServer":
         return t("patchbay.dest.addServer");
       case "server":
         return t("patchbay.dest.serverAt", { host: dest.host });
       case "custom":
         return t("patchbay.dest.endpointAt", { name: providerDisplayLabel("custom", t), host: dest.host });
       default:
         return providerDisplayLabel(dest.providerId, t);
     }
   }

   // A resolved route's destination in words, for status lines and the footer.
   export function routeDestinationLabel(route: ResolvedRoute, t: Translate): string {
     if (route.locality === "device") return t("locality.device");
     if (route.locality === "network") {
       return route.host ? t("patchbay.dest.serverAt", { host: route.host }) : t("locality.network");
     }
     if (route.providerId === "custom" && route.host) {
       return t("patchbay.dest.endpointAt", { name: providerDisplayLabel("custom", t), host: route.host });
     }
     return providerDisplayLabel(route.providerId, t);
   }

   export function privacyTally(routes: readonly ResolvedRoute[]) {
     let stay = 0;
     let network = 0;
     let remote = 0;
     for (const r of routes) {
       if (r.sends === null) stay++;
       else if (r.locality === "network") network++;
       else remote++;
     }
     return { total: routes.length, stay, network, remote };
   }

   export function cablePath(from: { x: number; y: number }, to: { x: number; y: number }): string {
     return `M ${from.x} ${from.y} C ${from.x + 120} ${from.y}, ${to.x - 120} ${to.y}, ${to.x} ${to.y}`;
   }
   ```

   ```ts
   import { useCallback, useMemo, useRef, useState } from "react";
   import type { ResolvedRoute, RouteCopy, RouteId, RouteSettingsSnapshot } from "../../../lib/routeGraph.ts";
   import type { RouteChangeOutcome } from "../../../stores/routeDraftStore.ts";
   import type { RouteTarget } from "../../../lib/routePlanner.ts";

     buildDestinations,
     checkCompatibility,
     compatContext,
     compatReason,
     destinationKeyOf,
     destinationKeyState,
     destinationLabel,
     routeDestinationLabel,
     targetForDestination,
     taskKind,
     type Compatibility,
     type KeyState,
     type RouteDestination,
   } from "./destinations.ts";

   type Translate = (key: string, options?: Record<string, unknown>) => string;

   export interface PatchbayBoardDeps {
     routes: ResolvedRoute[];
     snapshot: RouteSettingsSnapshot;
     sealed: boolean;
     providersFor: (id: RouteId) => readonly string[];
     request: (id: RouteId, target: RouteTarget) => Promise<RouteChangeOutcome | "staged">;
     copy: RouteCopy;
     t: Translate;
   }

   export interface PatchbayBoard {
     destinations: RouteDestination[];
     selectedId: RouteId | null;
     message: string | null;
     undo: (() => void) | null;
     compatOf: (dest: RouteDestination) => Compatibility | null;
     keyStateOf: (dest: RouteDestination) => KeyState | null;
     selectTask: (id: RouteId) => void;
     pickDestination: (dest: RouteDestination) => Promise<void>;
     clearSelection: () => boolean;
   }

   export function usePatchbayBoard(deps: PatchbayBoardDeps): PatchbayBoard {
     const { routes, snapshot, sealed, providersFor, request, copy, t } = deps;
     const [selectedId, setSelectedId] = useState<RouteId | null>(null);
     const [message, setMessage] = useState<string | null>(null);
     const [undo, setUndo] = useState<(() => void) | null>(null);
     const busy = useRef(false);

     const destinations = useMemo(() => buildDestinations(snapshot, routes), [snapshot, routes]);
     const ctx = useMemo(
       () => compatContext(snapshot, sealed, providersFor),
       [snapshot, sealed, providersFor]
     );
     const selected = routes.find((r) => r.id === selectedId) ?? null;
     const taskName = (id: RouteId) => t(`routes.names.${id}`);

     const compatOf = (dest: RouteDestination) =>
       selected ? checkCompatibility(selected.id, dest, ctx) : null;
     const keyStateOf = (dest: RouteDestination) =>
       selected ? destinationKeyState(dest, snapshot.facts, taskKind(selected.id)) : null;

     const selectTask = useCallback(
       (id: RouteId) => {
         const route = routes.find((r) => r.id === id);
         if (!route) return;
         setUndo(null);
         if (route.kind === "support") {
           setSelectedId(null);
           setMessage(t(`patchbay.locked.${id}`));
           return;
         }
         setMessage(null);
         setSelectedId((current) => (current === id ? null : id));
       },
       [routes, t]
     );

     const clearSelection = useCallback((): boolean => {
       if (selectedId === null) return false;
       setSelectedId(null);
       setMessage(null);
       return true;
     }, [selectedId]);

     const runUndo = (fn: () => RouteChangeOutcome | "staged", task: string) => {
       setUndo(null);
       const result = fn();
       if (result === "staged") setMessage(t("patchbay.hint.undoStaged"));
       else setMessage(result.applied ? t("patchbay.hint.undone", { task }) : t("patchbay.hint.notApplied"));
     };

     const announce = (route: ResolvedRoute, dest: RouteDestination, outcome: RouteChangeOutcome) => {
       const task = taskName(route.id);
       if (outcome.plan.kind === "noop") {
         setMessage(t("patchbay.hint.already", { task, destination: destinationLabel(dest, t) }));
         return;
       }
       if (!outcome.applied) {
         setMessage(t("patchbay.hint.notApplied"));
         return;
       }
       const own = outcome.plan.changes.find((c) => c.id === route.id);
       const destination = own ? routeDestinationLabel(own.after, t) : destinationLabel(dest, t);
       let text =
         route.state.status === "off"
           ? t("patchbay.hint.movedOff", { task, destination })
           : t("patchbay.hint.moved", { task, destination });
       const coupled = outcome.plan.changes.filter((c) => c.coupled);
       if (coupled.length > 0) {
         text += ` ${t("patchbay.hint.movedAlso", { tasks: copy.list(coupled.map((c) => taskName(c.id))) })}`;
       }
       setMessage(text);
       const undoFn = outcome.undo;
       if (undoFn) setUndo(() => () => runUndo(undoFn, task));
     };

     const pickDestination = useCallback(
       async (dest: RouteDestination) => {
         if (busy.current) return;
         if (!selected) {
           setMessage(t("patchbay.hint.pickTaskFirst"));
           return;
         }
         const compat = checkCompatibility(selected.id, dest, ctx);
         if (compat.ok === false) {
           setMessage(compatReason(compat, t));
           return;
         }
         const task = taskName(selected.id);
         if (destinationKeyOf(selected) === dest.id) {
           setMessage(
             selected.state.status === "off"
               ? t("patchbay.hint.off", { task })
               : t("patchbay.hint.already", { task, destination: destinationLabel(dest, t) })
           );
           return;
         }
         busy.current = true;
         setUndo(null);
         try {
           const outcome = await request(selected.id, targetForDestination(selected.id, dest));
           setSelectedId(null);
           if (outcome === "staged") {
             setMessage(t("patchbay.hint.staged", { task, destination: destinationLabel(dest, t) }));
           } else {
             announce(selected, dest, outcome);
           }
         } catch (error) {
           setMessage(t("patchbay.hint.failed", { error: (error as Error).message }));
         } finally {
           busy.current = false;
         }
       },
       // announce/runUndo only close over t, copy and setters
       // eslint-disable-next-line react-hooks/exhaustive-deps
       [selected, ctx, request, t, copy]
     );

     return {
       destinations,
       selectedId,
       message,
       undo,
       compatOf,
       keyStateOf,
       selectTask,
       pickDestination,
       clearSelection,
     };
   }
   ```

   ```tsx
   export function Patchbay({ onNavigateToPath }: { onNavigateToPath?: (path: string) => void }) {
     const { t, i18n } = useTranslation();
     const routes = useResolvedRoutes();
     const sealed = useSealedStore((s) => s.sealed);
     const draft = useRouteDraftStore((s) => s.draft);
     const [wide, setWide] = useState(true);
     const [cables, setCables] = useState<PatchbayCable[]>([]);
     const boardRef = useRef<HTMLDivElement>(null);
     const jacks = useRef(new Map<string, HTMLElement>());

     useEffect(() => {
       ensureSealedSubscribed();
     }, []);

     // eslint-disable-next-line react-hooks/exhaustive-deps
     const copy = useMemo(() => routeCopy(), [i18n.language]);
     // `routes` is rebuilt whenever settings, facts, sealed state or language
     // change, so the snapshot used for destinations is rebuilt with it.
     // eslint-disable-next-line react-hooks/exhaustive-deps
     const snapshot = useMemo(() => routeSnapshotFromSettings(useSettingsStore.getState()), [routes]);
     const request = useCallback(
       (id: RouteId, target: Parameters<typeof requestRouteChange>[1]) => requestRouteChange(id, target, copy),
       [copy]
     );
     const board = usePatchbayBoard({
       routes,
       snapshot,
       sealed,
       providersFor: providersForRoute,
       request,
       copy,
       t,
     });

     const registerJack = useCallback(
       (key: string) => (el: HTMLElement | null) => {
         if (el) jacks.current.set(key, el);
         else jacks.current.delete(key);
       },
       []
     );

     const measure = useCallback(() => {
       const el = boardRef.current;
       if (!el) return;
       const box = el.getBoundingClientRect();
       setWide(box.width >= 720);
       const centre = (key: string) => {
         const jack = jacks.current.get(key);
         if (!jack) return null;
         const r = jack.getBoundingClientRect();
         return { x: r.left + r.width / 2 - box.left, y: r.top + r.height / 2 - box.top };
       };
       const rtl = getComputedStyle(el).direction === "rtl";
       const next: PatchbayCable[] = [];
       for (const route of routes) {
         // A drafted route has no cable until Apply; an off route sends nothing.
         if (route.state.status === "off" || draft?.routeId === route.id) continue;
         const destId = destinationKeyOf(route);
         if (destId === null) continue;
         const from = centre(`task:${route.id}`);
         const to = centre(`dest:${destId}`);
         if (!from || !to) continue;
         const [a, b] = rtl ? [to, from] : [from, to];
         next.push({ id: route.id, d: cablePath(a, b), locality: route.locality, paused: route.state.status !== "ready" });
       }
       setCables(next);
     }, [routes, draft]);

     useLayoutEffect(() => {
       measure();
       const el = boardRef.current;
       if (!el || typeof ResizeObserver === "undefined") return;
       const observer = new ResizeObserver(() => measure());
       observer.observe(el);
       return () => observer.disconnect();
     }, [measure, board.destinations]);

     return (
       <PatchbayView
         routes={routes}
         destinations={board.destinations}
         destinationOf={destinationKeyOf}
         compatOf={board.compatOf}
         keyStateOf={board.keyStateOf}
         selectedId={board.selectedId}
         stagedRouteId={draft?.routeId ?? null}
         sealed={sealed}
         message={board.message}
         onUndo={board.undo}
         wide={wide}
         cables={cables}
         boardRef={boardRef}
         registerJack={registerJack}
         onSelectTask={board.selectTask}
         onPickDestination={(dest) => void board.pickDestination(dest)}
         onOpenSettings={(route) => onNavigateToPath?.(route.settingsPath)}
         onEscape={board.clearSelection}
       />
     );
   }
   ```
