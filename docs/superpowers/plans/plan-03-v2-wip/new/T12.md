**Why the rows come only from `keyring.list` (I11).** v1 wrote each Verify result straight into the row. A verdict that was computed for the old key could therefore land on a replacement. Main now binds each verification to a credential generation (Task 11) and may drop the result; this task never trusts a verify result for row content. After every Verify, every `keyring.onChanged` event and every reminder change the panel calls `list()` again, and an out-of-order response is discarded (the latest request wins). The panel reads only `outcome` from `verify`'s `{ outcome, entry }`: `superseded` shows "The key changed while it was being checked" instead of a verdict, `sealed` says sealed mode stopped it (and ends a Verify all run), `invalidOptions` names the Corti region and tenant that main refused, `missing` adds nothing (the removed credential's row disappears on the refresh that follows), and `recorded` adds nothing (the verdict arrives through `list()`).

```js
const tr = (key, options) => (options ? `${key}${JSON.stringify(options)}` : key);

const entry = (id, o = {}) => ({
  id, providerId: id.startsWith("corti:") ? "corti" : id,
  status: "unverified", lastVerifiedAt: null, lastErrorCode: null, lastErrorStatus: null, reminderDueAt: null, ...o,
});
const RECORDED = { outcome: "recorded", entry: null };
const defer = () => {
  let resolve;
  const promise = new Promise((r) => (resolve = r));
  return { promise, resolve };
};
const tick = () => React.act(async () => { await new Promise((r) => setImmediate(r)); });
function find(node, predicate) {
  if (Array.isArray(node)) return node.map((item) => find(item, predicate)).find(Boolean);
  if (!node || typeof node !== "object") return null;
  return predicate(node) ? node : find(node.props?.children, predicate);
}

function fakeKeyring(initial) {
  let rows = initial;
  const listeners = new Set();
  const api = {
    calls: { list: 0, verify: [], options: [], setReminder: [] },
    listeners,
    set: (next) => { rows = next; },
    emit: (id = "openai") => { for (const listener of [...listeners]) listener({ id }); },
    list: async () => { api.calls.list++; return rows.map((r) => ({ ...r })); },
    verify: (id, options) => { api.calls.verify.push(id); api.calls.options.push(options); return api.verifyImpl(id, options); },
    verifyImpl: async () => RECORDED,
    setReminder: async (id, enabled) => {
      api.calls.setReminder.push([id, enabled]);
      rows = rows.map((r) => (r.id === id ? { ...r, reminderDueAt: enabled ? 1 : null } : r));
      return rows.find((r) => r.id === id) ?? null;
    },
    onChanged: (cb) => { listeners.add(cb); return () => listeners.delete(cb); },
  };
  return api;
}
```

   ```ts
   import { useCallback, useEffect, useRef, useState } from "react";
   import type { CredentialId } from "../../lib/routes";

   // Main keeps a removed credential in memory for this long (Task 2). The
   // renderer never sees the value, only a token that can ask for it back.
   export const UNDO_WINDOW_MS = 8000;

   export type UndoFailureReason = "expired" | "replaced" | "unknown";

   export type RemovalNotice =
     | { kind: "removed"; token: string; expiresAt: number }
     | { kind: "undoFailed"; reason: UndoFailureReason }
     | { kind: "error"; message: string };

   // The subset of window.electronAPI.credentials this hook needs (Task 2).
   export interface CredentialsRemovalApi {
     remove(id: CredentialId): Promise<
       | { ok: true; undoToken: string; expiresAt: number; generation: number }
       | { ok: false; error: string }
     >;
     undoRemove(
       token: string
     ): Promise<{ ok: true; generation: number } | { ok: false; reason: UndoFailureReason }>;
   }

   export interface RemovalDeps {
     api?: CredentialsRemovalApi;
     now?: () => number;
     schedule?: (fn: () => void, ms: number) => unknown;
     cancel?: (handle: unknown) => void;
   }

   const defaultSchedule = (fn: () => void, ms: number): unknown => setTimeout(fn, ms);
   const defaultCancel = (handle: unknown) => clearTimeout(handle as ReturnType<typeof setTimeout>);

   export function useUndoableRemoval(credentialId: CredentialId | undefined, deps: RemovalDeps = {}) {
     const [notice, setNotice] = useState<RemovalNotice | null>(null);
     const [busy, setBusy] = useState(false);
     const timer = useRef<unknown>(null);
     const live = useRef(deps);
     live.current = deps;

     const resolveApi = (): CredentialsRemovalApi | undefined =>
       live.current.api ??
       (typeof window === "undefined"
         ? undefined
         : (window.electronAPI?.credentials as CredentialsRemovalApi | undefined));
     const available = !!credentialId && !!resolveApi()?.remove;

     const clearTimer = useCallback(() => {
       if (timer.current === null) return;
       (live.current.cancel ?? defaultCancel)(timer.current);
       timer.current = null;
     }, []);
     useEffect(() => clearTimer, [clearTimer]);

     const clear = useCallback(() => {
       clearTimer();
       setNotice(null);
     }, [clearTimer]);

     // `settle` is the field's own pending save: removing before main has
     // acknowledged it would let the late save resurrect the key.
     const remove = async (settle?: Promise<unknown> | null): Promise<void> => {
       const api = resolveApi();
       if (!credentialId || !api) return;
       clearTimer();
       setNotice(null);
       setBusy(true);
       try {
         if (settle) await settle;
         const result = await api.remove(credentialId);
         if (result.ok === false) {
           setNotice({ kind: "error", message: result.error });
           return;
         }
         const remaining = Math.max(
           0,
           Math.min(UNDO_WINDOW_MS, result.expiresAt - (live.current.now ?? Date.now)())
         );
         const token = result.undoToken;
         setNotice({ kind: "removed", token, expiresAt: result.expiresAt });
         timer.current = (live.current.schedule ?? defaultSchedule)(() => {
           timer.current = null;
           setNotice((current) =>
             current?.kind === "removed" && current.token === token ? null : current
           );
         }, remaining);
       } catch (error) {
         setNotice({ kind: "error", message: (error as Error).message });
       } finally {
         setBusy(false);
       }
     };

     const undo = async (): Promise<void> => {
       const api = resolveApi();
       if (!api || notice?.kind !== "removed") return;
       const { token } = notice;
       clearTimer();
       setBusy(true);
       try {
         const result = await api.undoRemove(token);
         setNotice(result.ok === true ? null : { kind: "undoFailed", reason: result.reason });
       } catch (error) {
         setNotice({ kind: "error", message: (error as Error).message });
       } finally {
         setBusy(false);
       }
     };

     return { available, notice, busy, remove, undo, clear };
   }
   ```

   ```tsx
   // computed (Task 11). The flag only explains why a row is not showing a
   // verdict; row content always comes from list().
   function isSuperseded(result: unknown): boolean {
     return (
       typeof result === "object" &&
       result !== null &&
       (result as { superseded?: unknown }).superseded === true
     );
   }

   export function useKeyringController(deps: {
     api: KeyringApi | undefined;
     flush: () => Promise<void>;
     getSealed: () => boolean;
   }): KeyringController {
     const [entries, setEntries] = useState<KeyringEntry[]>([]);
     const [loadError, setLoadError] = useState(false);
     const [checking, setChecking] = useState<CredentialId | null>(null);
     const [progress, setProgress] = useState<{ current: number; total: number } | null>(null);
     const [notices, setNotices] = useState<Partial<Record<CredentialId, KeyringNotice>>>({});
     const live = useRef(deps);
     live.current = deps;
     const entriesRef = useRef<KeyringEntry[]>([]);
     const requestSeq = useRef(0);
     const running = useRef(false);
     const alive = useRef(true);
     const api = deps.api;

     const refresh = useCallback(async () => {
       const current = live.current.api;
       if (!current) return;
       const seq = ++requestSeq.current;
       try {
         const next = await current.list();
         if (!alive.current || seq !== requestSeq.current) return; // the latest request wins
         entriesRef.current = next;
         setEntries(next);
         setLoadError(false);
       } catch {
         if (!alive.current || seq !== requestSeq.current) return;
         setLoadError(true); // keep the rows we already have
       }
     }, []);

     useEffect(() => {
       alive.current = true;
       void refresh();
       const unsubscribe = api?.onChanged?.(() => {
         void refresh();
       });
       return () => {
         alive.current = false;
         unsubscribe?.();
       };
     }, [api, refresh]);

     const setNotice = (id: CredentialId, notice: KeyringNotice | null) =>
       setNotices((current) => {
         const next = { ...current };
         if (notice) next[id] = notice;
         else delete next[id];
         return next;
       });

     // One credential: saves are acknowledged first, then the request goes out.
     const runVerify = async (id: CredentialId) => {
       setNotice(id, null);
       setChecking(id);
       try {
         await live.current.flush();
         const result = await live.current.api!.verify(id);
         if (isSuperseded(result)) setNotice(id, { kind: "superseded" });
       } catch (error) {
         setNotice(id, { kind: "failed", message: (error as Error).message });
       } finally {
         setChecking(null);
         await refresh();
       }
     };

     const verify = async (id: CredentialId) => {
       if (running.current || !live.current.api || live.current.getSealed()) return;
       running.current = true;
       try {
         await runVerify(id);
       } finally {
         running.current = false;
       }
     };

     const verifyAll = async () => {
       if (running.current || !live.current.api || live.current.getSealed()) return;
       running.current = true;
       try {
         const ids = entriesRef.current.map((e) => e.id);
         for (let i = 0; i < ids.length; i++) {
           // Sealed mode can turn on between requests; so can a removal.
           if (live.current.getSealed()) break;
           if (!entriesRef.current.some((e) => e.id === ids[i])) continue;
           setProgress({ current: i + 1, total: ids.length });
           await runVerify(ids[i]);
         }
       } finally {
         setProgress(null);
         running.current = false;
       }
     };

     const setReminder = async (id: CredentialId, enabled: boolean) => {
       try {
         await live.current.api?.setReminder(id, enabled);
       } catch (error) {
         setNotice(id, { kind: "failed", message: (error as Error).message });
       } finally {
         await refresh();
       }
     };

     return { entries, loadError, checking, progress, notices, verify, verifyAll, setReminder };
   }
   ```
