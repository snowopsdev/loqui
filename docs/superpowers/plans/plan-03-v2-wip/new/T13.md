**Commits stay on Task 9's path.** This task changes only how local models are shown. Task 9 runs first and already sends every local pick through the planner: `TranscriptionModelPicker`'s `handleWhisperModelSelect` and `handleParakeetModelSelect`, and `ReasoningModelSelector`'s `handleModelSelect` (the `onModelSelect` it passes to `LocalModelPicker`), call `onCommit(target)`. The grids call those same handlers, including the auto-select after a download and the imported-model "test load then select". This task adds no second commit path and no route prop.

Local models become **cartridges**: a 2-column grid of cards, each showing:
- the name;
- three 5-pip meters (Speed, Quality, Memory use), always labelled in words;
- size and context length;
- the state: Ready, Download, Downloading with a real `role="progressbar"`, or Too large for this device;
- "In use by" chips that show which tasks already run on the model.

The grid is a `role="radiogroup"` with one tab stop:
- Arrows, Home and End move **focus** (←/→ swap in RTL, ↑/↓ move a row). They never choose a model, because choosing changes a route. This matches Plan 02's SET-Q2 ("moving focus must never switch where audio goes") and Task 5's mode selector.
- Enter, Space or a click chooses a downloaded card. An undownloaded card is `aria-disabled` and stays focusable, so a screen-reader user finds it next to its Download button. **A download starts only from that button.** This deliberately narrows the v1 text ("Enter starts the download"): a stray keypress should never begin a multi-gigabyte download, and today's cards behave the same way.
- Each card's action button (Download, Cancel or Delete) and its model-card link are tabbable only on the card that holds the tab stop, so Tab goes radio, then that card's controls, then out. Arrows pressed on those controls move to the next card's radio.
- Sorting is Recommended first, then by size ascending, collapsed after 5 with "Show all 13" (Q8). The toggle stays in place and becomes "Show fewer", so focus is never lost.
- Download buttons use the outline variant except on the Recommended card (Q9).

This supersedes Plan 02's SET-Q2 markup for local models only; cloud model lists keep `ModelCardList`.

Pips are estimates derived from size unless the registry carries curated `speedTier` and `qualityTier`; the caption "estimate" shows only while a visible meter is size-derived. No card is dimmed with `opacity-*`: at 70% muted text on white is 3.6:1, and the reason a model is too large must stay readable.

```js
function walk(node, visit) {
  if (Array.isArray(node)) return node.forEach((child) => walk(child, visit));
  if (!node || typeof node !== "object") return;
  visit(node);
  walk(node.props?.children, visit);
```

   ```ts
   // Grid navigation over `count` items laid out `columns` per row, in reading
   // order. Left and right follow the reading direction, so they swap when `rtl`
   // is true; up and down move by a row and stop at the ends rather than wrapping.
   export function stepGrid(
     index: number,
     key: string,
     opts: { columns: number; count: number; rtl: boolean }
   ): number {
     const { columns, count, rtl } = opts;
     if (count === 0) return index;
     const last = count - 1;
     const forward = rtl ? "ArrowLeft" : "ArrowRight";
     const backward = rtl ? "ArrowRight" : "ArrowLeft";
     switch (key) {
       case forward:
         return Math.min(last, index + 1);
       case backward:
         return Math.max(0, index - 1);
       case "ArrowDown":
         return Math.min(last, index + columns);
       case "ArrowUp":
         return index - columns >= 0 ? index - columns : index;
       case "Home":
         return 0;
       case "End":
         return last;
       default:
         return index;
     }
   }
   ```

2. Create `src/components/ui/cartridgeMath.ts`:

   ```ts
   // Pure helpers for model cartridges (SET-R4 B). Tiers are estimates from file
   // size unless the registry carries curated speedTier/qualityTier.

   export interface CartridgeModel {
     id: string;
     name: string;
     sizeBytes?: number;
     contextLength?: number;
     recommended?: boolean;
     isDownloaded: boolean;
     isDownloading: boolean;
     isCancelling?: boolean;
     /** The archive is downloaded and unpacking: cancelling is no longer possible. */
     isInstalling?: boolean;
     progress?: number;
     speedTier?: number;
     qualityTier?: number;
     /** Short extra fact, already translated (for example an import's load status). */
     badge?: string;
     /** Model card or documentation link, with its already-translated label. */
     href?: string;
     hrefLabel?: string;
   }

   export type Fit = "comfortable" | "tight" | "tooLarge" | "unknown";

   const GB = 1_000_000_000;
   const bucket = (bytes: number, edges: number[]) => {
     const i = edges.findIndex((edge) => bytes <= edge * GB);
     return i === -1 ? edges.length + 1 : i + 1;
   };

   export function estimateTiers(m: Pick<CartridgeModel, "sizeBytes" | "speedTier" | "qualityTier">) {
     const size = m.sizeBytes;
     const memory = size ? bucket(size, [1.5, 3, 5, 8]) : null;
     const sizeSpeed = size ? 6 - bucket(size, [1, 2, 4, 8]) : null;
     const sizeQuality = size ? Math.min(5, bucket(size, [1, 2, 4, 8]) + 1) : null;
     const curated = m.speedTier !== undefined && m.qualityTier !== undefined;
     return {
       speed: m.speedTier ?? sizeSpeed,
       quality: m.qualityTier ?? sizeQuality,
       memory,
       estimated: !curated,
     };
   }

   export function estimateRamBytes(sizeBytes: number): number {
     return sizeBytes * 1.15;
   }

   export function fitFor(sizeBytes: number | undefined, totalMemoryBytes: number | null): Fit {
     if (!sizeBytes || !totalMemoryBytes) return "unknown";
     const share = estimateRamBytes(sizeBytes) / totalMemoryBytes;
     if (share < 0.4) return "comfortable";
     return share <= 0.7 ? "tight" : "tooLarge";
   }

   export function sortForDisplay<T extends Pick<CartridgeModel, "recommended" | "sizeBytes">>(
     models: T[]
   ): T[] {
     return models
       .map((m, i) => ({ m, i }))
       .sort((a, b) => {
         if (!!a.m.recommended !== !!b.m.recommended) return a.m.recommended ? -1 : 1;
         const sa = a.m.sizeBytes ?? Number.POSITIVE_INFINITY;
         const sb = b.m.sizeBytes ?? Number.POSITIVE_INFINITY;
         return sa === sb ? a.i - b.i : sa - sb;
       })
       .map(({ m }) => m);
   }

   export function visibleModels<T extends Pick<CartridgeModel, "id">>(
     sorted: T[],
     selectedId: string,
     expanded: boolean,
     limit = 5
   ): T[] {
     if (expanded || sorted.length <= limit) return sorted;
     const head = sorted.slice(0, limit);
     const selected = sorted.find((m) => m.id === selectedId);
     return selected && !head.includes(selected) ? [...head, selected] : head;
   }

     id: string;
     name: string;
     sizeBytes?: number;
     contextLength?: number;
     recommended?: boolean;
     speedTier?: number;
     qualityTier?: number;
   }

   export function cartridgeFromLocalModel(

     };
   }

   // Which tasks already run on each model: device routes that are not turned off,
   // matched by display name or id (the resolver reports the model it resolved).
   export function usedByMap(
     models: readonly Pick<CartridgeModel, "id" | "name">[],
     routes: readonly {
       id: string;
       locality: string;
       model: string | null;
       state: { status: string };
     }[],
     taskName: (routeId: string) => string
   ): Record<string, string[]> {
     const map: Record<string, string[]> = {};
     for (const model of models) {
       const tasks = routes
         .filter(
           (route) =>
             route.locality === "device" &&
             route.state.status !== "off" &&
             (route.model === model.name || route.model === model.id)
         )
         .map((route) => taskName(route.id));
       if (tasks.length) map[model.id] = tasks;
     }
     return map;
   }
   ```

     ```ts
     // Total system memory in bytes, for the "too large for this device" check on
     // model cartridges. null until main answers, and null if it cannot (browser

     export function useTotalMemory(): number | null {

       useEffect(() => {
         let mounted = true;
         const read = window.electronAPI?.getTotalMemory?.();
         if (read) {
           read.then(

             },
             () => {

             }

         }
         return () => {
           mounted = false;
         };
       }, []);

     }
     ```
