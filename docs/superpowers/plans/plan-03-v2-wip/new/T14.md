   // Main-owned storage counts and deletions for Settings > Privacy & Data.
   // Each preview counts with the predicate its deletion uses, and each
   // deletion refuses to run when the set changed after the user reviewed it,
   // so a confirmation never promises less than the delete removes.
   // Unreadable state is an error, never zero.

   /** Audio predicate: every recording in the audio folder, linked to a transcript or not. */
   function isStoredAudioFile(name) {
     return typeof name === "string" && name.endsWith(".webm");
   }

   // Recordings end in "-<transcription id>.webm" (audioStorage.js _buildFilename).
   const AUDIO_ID = /(?:^|-)(\d+)\.webm$/;

   /**
    * History predicate: every row of `transcriptions` -- completed, discarded
    * and soft-deleted alike. DatabaseManager.clearTranscriptions deletes the
    * same set (`DELETE FROM transcriptions`, database.js:553); deleteHistory
    * compares the two inside one transaction and rolls back on any mismatch.
    */
   const HISTORY_COUNT_SQL = `SELECT
     COUNT(*) AS total,
     COALESCE(SUM(CASE WHEN deleted_at IS NULL AND status = 'discarded' THEN 1 ELSE 0 END), 0) AS discarded,
     COALESCE(SUM(CASE WHEN deleted_at IS NOT NULL THEN 1 ELSE 0 END), 0) AS softDeleted,
     COALESCE(SUM(CASE WHEN has_audio = 1 THEN 1 ELSE 0 END), 0) AS withAudio
   FROM transcriptions`;

   const describe = (error) => String(error?.message || error);
   const isCount = (value) => Number.isInteger(value) && value >= 0;
   const validExpected = (expected, keys) =>
     !!expected && typeof expected === "object" && keys.every((key) => isCount(expected[key]));

   class StaleHistoryError extends Error {
     constructor(total) {
       super("stale");
       this.total = total;
     }
   }

   function createStorageOps({ audio, database, broadcast = () => {} }) {
     const countHistory = () => {
       const row = database.db.prepare(HISTORY_COUNT_SQL).get();
       return {
         total: row.total,
         discarded: row.discarded,
         softDeleted: row.softDeleted,
         withAudio: row.withAudio,
       };
     };

     function preview() {
       let audioPreview;
       try {
         const measured = audio.measure();
         audioPreview = {
           files: measured.files.length,
           bytes: measured.bytes,
           directory: measured.directory,
         };
       } catch (error) {
         audioPreview = { error: describe(error) };
       }
       let history;
       try {
         history = countHistory();
       } catch (error) {
         history = { error: describe(error) };
       }
       return { audio: audioPreview, history };
     }

     // Unlinks exactly the counted names, then re-reads the folder to learn how
     // many of them are still there. A failed re-read leaves `remaining` unknown.
     function removeCounted(names) {
       const errors = [];
       const { deleted, failed } = audio.removeFiles(names);
       for (const failure of failed) errors.push(`delete-failed: ${failure.file}: ${failure.error}`);
       let present = null;
       let remaining = null;
       try {
         present = new Set(audio.measure().files);
         remaining = names.filter((name) => present.has(name)).length;
       } catch (error) {
         errors.push(`recount-failed: ${describe(error)}`);
       }
       return { deleted, remaining, present, errors };
     }

     // Clears has_audio only for rows whose recording is provably gone.
     function reconcileAudioFlags(present, errors) {
       if (!present) return;
       try {
         const kept = new Set();
         for (const name of present) {
           const match = AUDIO_ID.exec(name);
           if (match) kept.add(Number(match[1]));
         }
         const gone = database.db
           .prepare("SELECT id FROM transcriptions WHERE has_audio = 1")
           .all()
           .map((row) => row.id)
           .filter((id) => !kept.has(id));
         if (gone.length > 0) database.clearAudioFlags(gone);
       } catch (error) {
         errors.push(`flags-failed: ${describe(error)}`);
       }
     }

     function deleteAudio({ expected, reconcileFlags = true } = {}) {
       if (expected !== undefined && !validExpected(expected, ["files"])) {
         return { ok: false, deleted: 0, remaining: null, errors: ["invalid-request"] };
       }
       let counted;
       try {
         counted = audio.measure();
       } catch (error) {
         return { ok: false, deleted: 0, remaining: null, errors: [`count-failed: ${describe(error)}`] };
       }
       if (expected !== undefined && counted.files.length !== expected.files) {
         return { ok: false, deleted: 0, remaining: counted.files.length, errors: ["stale"] };
       }
       const result = removeCounted(counted.files);
       if (reconcileFlags) reconcileAudioFlags(result.present, result.errors);
       return {
         ok: result.errors.length === 0 && result.remaining === 0,
         deleted: result.deleted,
         remaining: result.remaining,
         errors: result.errors,
       };
     }

     function deleteHistory({ expected } = {}) {
       const untouched = (files) => ({ deleted: 0, remaining: files });
       if (expected !== undefined && !validExpected(expected, ["total", "files"])) {
         return { ok: false, deleted: 0, remaining: null, errors: ["invalid-request"], audio: untouched(null) };
       }
       let counted;
       try {
         counted = audio.measure();
       } catch (error) {
         return {
           ok: false,
           deleted: 0,
           remaining: null,
           errors: [`count-failed: ${describe(error)}`],
           audio: untouched(null),
         };
       }
       // Rows first, in one transaction: a stale or drifting count rolls back
       // before any recording is touched.
       let cleared;
       try {
         cleared = database.db.transaction(() => {
           const rows = countHistory();
           if (
             expected !== undefined &&
             (rows.total !== expected.total || counted.files.length !== expected.files)
           ) {
             throw new StaleHistoryError(rows.total);
           }
           const result = database.clearTranscriptions();
           if (result?.cleared !== rows.total) {
             throw new Error(
               `predicate mismatch: counted ${rows.total} rows, delete removed ${result?.cleared}`
             );
           }
           return result.cleared;
         })();
       } catch (error) {
         if (error instanceof StaleHistoryError) {
           return { ok: false, deleted: 0, remaining: error.total, errors: ["stale"], audio: untouched(counted.files.length) };
         }
         let remaining = null;
         try {
           remaining = countHistory().total;
         } catch {
           // the database itself is unreadable; remaining stays unknown
         }
         return {
           ok: false,
           deleted: 0,
           remaining,
           errors: [`history-failed: ${describe(error)}`],
           audio: untouched(counted.files.length),
         };
       }
       broadcast("transcriptions-cleared", { cleared });
       broadcast("analytics-changed");
       // Every row is gone, so no has_audio flag is left to reconcile.
       const files = removeCounted(counted.files);
       return {
         ok: files.errors.length === 0 && files.remaining === 0,
         deleted: cleared,
         remaining: 0,
         errors: files.errors,
         audio: { deleted: files.deleted, remaining: files.remaining },
       };
     }

     return { preview, deleteAudio, deleteHistory };
   }

   module.exports = { createStorageOps, isStoredAudioFile, HISTORY_COUNT_SQL };

   // (src/helpers/storageOps.js); this module validates what main returns and
   // does the renderer-side model arithmetic. Nothing here turns an unknown
   // into zero or a failure into success.

   export type ModelFamily = "whisper" | "parakeet" | "llm";
   export const MODEL_FAMILIES: readonly ModelFamily[] = ["whisper", "parakeet", "llm"];

   export interface AudioSnapshot { kind: "audio"; files: number; bytes: number; directory: string }
   export interface HistorySnapshot {
     kind: "history";
     total: number;
     discarded: number;
     softDeleted: number;
     files: number;
     bytes: number;
     directory: string;
   }
   export interface ModelsSnapshot { kind: "models"; count: number; bytes: number; directory: string | null }
   export type StorageSnapshot = AudioSnapshot | HistorySnapshot | ModelsSnapshot;
   export type StorageKind = StorageSnapshot["kind"];
   export type RowState<S extends StorageSnapshot = StorageSnapshot> =
     | { status: "loading" }
     | { status: "error"; error: string }
     | { status: "ready"; snapshot: S };

   export interface FamilyOutcome { family: ModelFamily; ok: boolean; error?: string }
   export interface DeleteOutcome {
     ok: boolean;
     deleted: number | null; // null: main's answer is unknown
     remaining: number | null; // null: could not be recounted
     errors: string[];
     audio?: { deleted: number | null; remaining: number | null }; // history only
     families?: FamilyOutcome[]; // models only
   }

   interface SpeechModelInfo { downloaded?: boolean; size_bytes?: number; size_mb?: number }
   interface LlmModelInfo { isDownloaded?: boolean; sizeBytes?: number; downloadedSize?: number }
   export interface ModelLists {
     whisper?: { models?: SpeechModelInfo[] };
     parakeet?: { models?: SpeechModelInfo[] };
     llm?: LlmModelInfo[];
   }

   type Loose = Record<string, unknown>;
   const isObject = (value: unknown): value is Loose => !!value && typeof value === "object";
   const isCount = (value: unknown): value is number =>
     typeof value === "number" && Number.isInteger(value) && value >= 0;

   export function errorText(error: unknown): string {
     if (error instanceof Error) return error.message;
     if (isObject(error) && typeof error.error === "string") return error.error;
     return String(error ?? "unknown error");
   }

   export function rowsFromPreview(raw: unknown): {
     audio: RowState<AudioSnapshot>;
     history: RowState<HistorySnapshot>;
   } {
     const audioRaw = isObject(raw) ? raw.audio : undefined;
     const historyRaw = isObject(raw) ? raw.history : undefined;

     let audio: RowState<AudioSnapshot> = { status: "error", error: "invalid-preview" };
     if (isObject(audioRaw) && typeof audioRaw.error === "string") {
       audio = { status: "error", error: audioRaw.error };
     } else if (isObject(audioRaw)) {
       const { files, bytes, directory } = audioRaw;
       if (isCount(files) && isCount(bytes) && typeof directory === "string") {
         audio = { status: "ready", snapshot: { kind: "audio", files, bytes, directory } };
       }
     }

     let history: RowState<HistorySnapshot> = { status: "error", error: "invalid-preview" };
     if (isObject(historyRaw) && typeof historyRaw.error === "string") {
       history = { status: "error", error: historyRaw.error };
     } else if (audio.status === "error") {
       // History deletion also deletes every recording, so it can't be counted without them.
       history = { status: "error", error: `audio: ${audio.error}` };
     } else if (isObject(historyRaw) && audio.status === "ready") {
       const { total, discarded, softDeleted } = historyRaw;
       if (isCount(total) && isCount(discarded) && isCount(softDeleted)) {
         const { files, bytes, directory } = audio.snapshot;
         history = {
           status: "ready",
           snapshot: { kind: "history", total, discarded, softDeleted, files, bytes, directory },
         };
       }
     }
     return { audio, history };
   }

   const MIB = 1024 * 1024;

   export function summarizeModels(lists: ModelLists): { count: number; bytes: number } {
     let count = 0;
     let bytes = 0;
     for (const m of [...(lists.whisper?.models ?? []), ...(lists.parakeet?.models ?? [])]) {
       if (!m.downloaded) continue;
       count++;
       // Main reports exact bytes; size_mb is rounded MiB (whisper.js:714-715, parakeet.js:625-626).
       bytes += m.size_bytes ?? (m.size_mb ?? 0) * MIB;
     }
     for (const m of lists.llm ?? []) {
       if (!m.isDownloaded) continue;
       count++;
       bytes += m.downloadedSize || m.sizeBytes || 0;
     }
     return { count, bytes };
   }

   export function countModels(
     settled: Record<ModelFamily, PromiseSettledResult<unknown>>
   ): { ok: true; count: number; bytes: number } | { ok: false; error: string } {
     const errors: string[] = [];
     const read = (family: ModelFamily): unknown => {
       const result = settled[family];
       if (result.status === "rejected") {
         errors.push(`${family}: ${errorText(result.reason)}`);
         return undefined;
       }
       const value = result.value;
       const valid =
         family === "llm"
           ? Array.isArray(value)
           : isObject(value) && value.success === true && Array.isArray(value.models);
       if (!valid) {
         const reason = isObject(value) && typeof value.error === "string" ? value.error : "unreadable model list";
         errors.push(`${family}: ${reason}`);
         return undefined;
       }
       return value;
     };
     const whisper = read("whisper") as ModelLists["whisper"];
     const parakeet = read("parakeet") as ModelLists["parakeet"];
     const llm = read("llm") as ModelLists["llm"];
     if (errors.length > 0) return { ok: false, error: errors.join("; ") };
     return { ok: true, ...summarizeModels({ whisper, parakeet, llm }) };
   }

   export function familyOutcome(family: ModelFamily, settled: PromiseSettledResult<unknown>): FamilyOutcome {
     if (settled.status === "rejected") return { family, ok: false, error: errorText(settled.reason) };
     const value = settled.value;
     if (isObject(value) && value.success === true) return { family, ok: true };
     const error = isObject(value) && typeof value.error === "string" ? value.error : "no success result";
     return { family, ok: false, error };
   }

   export function modelsOutcome(
     before: number,
     families: FamilyOutcome[],
     remaining: number | null,
     recountError: string | null
   ): DeleteOutcome {
     const errors = families
       .filter((f) => f.ok === false)
       .map((f) => `${f.family}: ${f.error ?? "failed"}`);
     if (recountError !== null) errors.push(`recount-failed: ${recountError}`);
     return {
       ok: errors.length === 0 && remaining === 0,
       deleted: remaining === null ? null : Math.max(0, before - remaining),
       remaining,
       errors,
       families,
     };
   }

   function readResult(raw: unknown): DeleteOutcome | null {
     if (!isObject(raw)) return null;
     const { ok, deleted, remaining, errors } = raw;
     if (typeof ok !== "boolean" || !isCount(deleted) || !Array.isArray(errors)) return null;
     if (remaining !== null && !isCount(remaining)) return null;
     const list = errors.map(String);
     return {
       ok: ok === true && list.length === 0 && remaining === 0,
       deleted,
       remaining: remaining as number | null,
       errors: list,
     };
   }

   const INVALID: DeleteOutcome = { ok: false, deleted: null, remaining: null, errors: ["invalid-result"] };

   export function outcomeFromResult(raw: unknown): DeleteOutcome {
     return readResult(raw) ?? { ...INVALID, errors: [...INVALID.errors] };
   }

   export function historyOutcomeFromResult(raw: unknown): DeleteOutcome {
     const rows = readResult(raw);
     const audio = isObject(raw) && isObject(raw.audio) ? raw.audio : null;
     if (!rows || !audio || !isCount(audio.deleted)) {
       return { ...INVALID, errors: [...INVALID.errors], audio: { deleted: null, remaining: null } };
     }
     const remaining = isCount(audio.remaining) ? audio.remaining : null;
     return {
       ...rows,
       ok: rows.ok === true && remaining === 0,
       audio: { deleted: audio.deleted, remaining },
     };
   }

     const outcome: DeleteOutcome = {
       ok: false,
       deleted: null,
       remaining: null,
       errors: [`ipc-failed: ${errorText(error)}`],

     if (!isObject(raw)) return { ok: false, errors: ["reset-failed: no result"] };
     const errors = Array.isArray(raw.errors) ? raw.errors.map(String) : [];
     if (raw.success === true && errors.length === 0) return { ok: true, errors: [] };
     if (errors.length > 0) return { ok: false, errors };
     return {
       ok: false,
       errors: [`reset-failed: ${typeof raw.message === "string" ? raw.message : "unknown"}`],
     };
   }

   export function resetConfirmed(typed: string, word: string): boolean {
     const expected = word.trim().toLocaleLowerCase();
     return expected.length > 0 && typed.trim().toLocaleLowerCase() === expected;
   }

   export function snapshotCount(snapshot: StorageSnapshot): number {
     if (snapshot.kind === "audio") return snapshot.files;
     if (snapshot.kind === "history") return snapshot.total;
     return snapshot.count;
   }

   export function isEmptySnapshot(snapshot: StorageSnapshot): boolean {
     if (snapshot.kind === "history") return snapshot.total === 0 && snapshot.files === 0;
     return snapshotCount(snapshot) === 0;
   }

   async function performDelete(api: Api, s: StorageSnapshot): Promise<DeleteOutcome> {
     try {
       if (s.kind === "models") return await deleteModels(api, s);
       const storage = api?.storage;
       if (!storage) throw new Error("unavailable in this window");
       if (s.kind === "audio") return outcomeFromResult(await storage.deleteAudio({ files: s.files }));
       return historyOutcomeFromResult(await storage.deleteHistory({ total: s.total, files: s.files }));

   export function StorageList() {
     const { t } = useTranslation();

       const api = window.electronAPI;

       const next: Rows = { audio: preview.audio, history: preview.history, models };

       return next;
     }, []);

       void load();
     }, [load]);

       if (!dialog || dialog.phase !== "review" || busy.current) return;
       busy.current = true;

       setDialog({ ...dialog, phase: "running" });
       const outcome = await performDelete(window.electronAPI, reviewed);
       if (reviewed.kind === "models") {

         // is gone; after a partial delete the survivors stay selectable and

         window.dispatchEvent(new Event("openwhispr-models-cleared"));
       }
       const fresh = await load(); // refreshes the rows; never decides the outcome
       busy.current = false;
       if (outcome.ok === true) {
         setDialog(null);
         toast({ title: doneTitle(t, reviewed, outcome), variant: "success", duration: 4500 });
         return;
       }
       const last = { snapshot: reviewed, outcome };
       const row = fresh[reviewed.kind];
       if (outcome.errors.includes("stale") && row.status === "ready") {
         setDialog({ snapshot: row.snapshot, phase: "review", last });
       } else {
         setDialog({ snapshot: reviewed, phase: "failed", last });
       }
     };

     const recount = async () => {
       if (!dialog || dialog.phase !== "failed") return;
       const current = dialog;
       const row = (await load())[current.snapshot.kind];
       if (row.status === "ready") {
         setDialog({ snapshot: row.snapshot, phase: "review", last: current.last });
       } else if (row.status === "error" && current.last) {
         const outcome = { ...current.last.outcome, errors: [...current.last.outcome.errors, `count-failed: ${row.error}`] };
         setDialog({ ...current, last: { ...current.last, outcome } });
       }

       if (!reset || reset.phase === "running" || reset.phase === "done") return;
       if (!resetConfirmed(reset.typed, word)) return;

       setReset({ phase: "running", typed, errors: [] });

       } catch (error) {
         result = { ok: false, errors: [`ipc-failed: ${errorText(error)}`] };
       }
       if (result.ok === false) {
         setReset({ phase: "failed", typed, errors: result.errors });
         return;
       }
       setReset({ phase: "done", typed, errors: [] });
       setTimeout(() => void window.electronAPI?.relaunchApp?.(), RESET_RELAUNCH_DELAY_MS);
     };

     const closeDialog = () => {
       if (dialog && dialog.phase !== "running") setDialog(null);
     };

       if (reset && reset.phase !== "running" && reset.phase !== "done") setReset(null);
     };

           models={rows.models}
           history={rows.history}

             const row = rows[kind];

               setDialog({ snapshot: row.snapshot, phase: "review", last: null });
             }
           }}
           onOpenModels={() => void window.electronAPI?.openWhisperModelsFolder?.()}

             phase={reset.phase}
             typed={reset.typed}

             }
             onTypedChange={(typed) =>
               setReset((current) => (current && current.phase !== "running" ? { ...current, typed } : current))
             }
             onConfirm={() => void runReset()}
             onCancel={closeReset}
             onOpenChange={(open) => {

             }}
           />
