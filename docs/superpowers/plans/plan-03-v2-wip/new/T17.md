   ```ts
   import { providerDisplayLabel } from "../../lib/locality";
   import type { ResolvedRoute } from "../../lib/routes";

   type Translate = (key: string, options?: Record<string, unknown>) => string;

   // SET-Q15: name where the screenshot goes and what rides with it, from the assistantVision route.
   // `sends` is Task 3's translated payload list and already includes the screenshot.
   export function screenContextCopy(
     route: Pick<ResolvedRoute, "locality" | "providerId" | "host" | "model" | "sends">,
     t: Translate
   ): string {
     if (route.locality === "device") {
       return route.model
         ? t("dictationAgent.screenContext.goesToDevice", { model: route.model })
         : t("dictationAgent.screenContext.goesToDeviceNoModel");
     }
     const payload = route.sends ?? t("routes.payload.screenshot");
     if (route.locality === "network") {
       return route.host
         ? t("dictationAgent.screenContext.goesToServer", { host: route.host, payload })
         : t("dictationAgent.screenContext.goesToServerNoHost", { payload });
     }
     const provider = providerDisplayLabel(route.providerId, (key) => t(key));
     return route.host
       ? t("dictationAgent.screenContext.goesToProvider", { provider, host: route.host, payload })
       : t("dictationAgent.screenContext.goesToProviderNoHost", { provider, payload });
   }
   ```

   ```ts
   import { useCallback, useEffect } from "react";
   import { useLocalStorage } from "./useLocalStorage";

   export const AUTO_LEARN_CHANGED_EVENT = "loqui-auto-learn-changed";

   export function useAutoLearnCorrections() {
     const [enabled, setEnabledRaw] = useLocalStorage("autoLearnCorrections", true, {
       serialize: String,
       deserialize: (value: string) => value !== "false",
     });

     const setEnabled = useCallback(
       (next: boolean) => {
         setEnabledRaw(next);
         window.electronAPI?.setAutoLearnEnabled?.(next);
         window.dispatchEvent(new CustomEvent(AUTO_LEARN_CHANGED_EVENT, { detail: next }));
       },
       [setEnabledRaw]
     );

     // Settings → Dictation and the Dictionary header each mount this hook.
     useEffect(() => {
       const onChanged = (event: Event) => setEnabledRaw((event as CustomEvent<boolean>).detail);
       window.addEventListener(AUTO_LEARN_CHANGED_EVENT, onChanged);
       return () => window.removeEventListener(AUTO_LEARN_CHANGED_EVENT, onChanged);
     }, [setEnabledRaw]);

     // Sync the stored state to the main process on mount.
     useEffect(() => {
       window.electronAPI?.setAutoLearnEnabled?.(enabled);
       // eslint-disable-next-line react-hooks/exhaustive-deps
     }, []);

     return [enabled, setEnabled] as const;
   }
   ```
